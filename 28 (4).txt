const CustomSettingsInput: React.FC<{
  value: string;
  onChange: (v: string) => void;
  placeholder: string;
  onBlur?: () => void;
}> = ({ value, onChange, placeholder, onBlur }) => {
  const [isFocused, setIsFocused] = useState(false);
  const ref = useRef<HTMLTextAreaElement>(null);
  useEffect(() => {
    if (ref.current) {
      if (isFocused) {
        ref.current.style.height = 'auto';
        ref.current.style.height = Math.min(ref.current.scrollHeight, 120) + 'px';
      } else {
        ref.current.style.height = '40px';
      }
    }
  }, [value, isFocused]);
  return (
    <textarea
      ref={ref}
      value={value}
      onChange={(e) => onChange(e.target.value)}
      onFocus={() => setIsFocused(true)}
      onBlur={() => { setIsFocused(false); onBlur?.(); }}
      placeholder={placeholder}
      className="w-full bg-[var(--ui-input)] border border-[var(--ui-border)] rounded-lg px-3 py-2 text-xs outline-none focus:border-[var(--ui-accent)] transition-all resize-none mt-1.5 font-medium placeholder:text-[var(--ui-text-muted)]"
      style={{ height: '40px', maxHeight: '120px' }}
    />
  );
};

                  </div>
                  <div className="space-y-1">
                    <label className="text-[10px] uppercase font-bold text-[var(--ui-text-muted)]">ภาษาพูด</label>
                    <select value={spokenLanguage} onChange={e => onUpdate({ ...state.script, spokenLanguage: e.target.value })} className="w-full bg-[var(--ui-input)] border border-[var(--ui-border)] rounded-lg px-2 h-9 text-xs outline-none">{SPOKEN_LANGUAGES.map(sl => <option key={sl} value={sl}>{sl}</option>)}</select>
                    {spokenLanguage === 'กำหนดเอง (Custom)' && <CustomSettingsInput value={customSpokenLanguage || ''} onChange={v => onUpdate({ ...state.script, customSpokenLanguage: v, customSpokenLanguageEn: undefined })} placeholder="เช่น ไทยปนอังกฤษ, ลาว" />}
                  </div>
                </div>
                <div className="space-y-1"><label className="text-[10px] uppercase font-bold text-[var(--ui-text-muted)]">โมเดลประมวลผล</label><select value={state.config.textModel} onChange={e => onConfigUpdate({ ...state.config, textModel: e.target.value as any })} className="w-full bg-[var(--ui-input)] border border-[var(--ui-border)] rounded-lg px-2 h-9 text-xs outline-none">{TEXT_MODELS.map(m => <option key={m.value} value={m.value}>{m.label}</option>)}</select></div>
              </div>
            )}
          </div>
          <div className="space-y-3">
            <div className="flex flex-col gap-3">
              <div className="flex gap-3">
                <button onClick={handleProcessClick} disabled={loading || stopping || !original || (scriptMode !== 'user' && !timeValidation.valid)} className="flex-[3] h-14 font-black rounded-xl transition-all flex items-center justify-center gap-2 shadow-lg disabled:opacity-30 disabled:cursor-not-allowed g2-cta">
                  {loading ? <div className="w-5 h-5 border-2 border-current border-t-transparent rounded-full animate-spin" /> : <span className="material-symbols-outlined font-bold">{stopping ? 'sync_disabled' : (scriptMode === 'user' ? 'person_search' : 'auto_awesome')}</span>}
                  {stopping ? 'กำลังหยุดงานเดิม...' : (scriptMode === 'user' ? 'ใช้บทนี้ + ดึงตัวละครและฉาก' : 'เขียนบทด้วย AI')}
                </button>
                <button onClick={handleUndo} disabled={loading || history.length <= 1} className="flex-1 h-14 bg-[var(--ui-surface-soft)] border border-[var(--ui-border)] text-[var(--ui-text-muted)] rounded-xl hover:text-[var(--ui-text-primary)] disabled:opacity-20 flex items-center justify-center"><span className="material-symbols-outlined font-bold">undo</span></button>
              </div>
              {scriptMode !== 'user' && !timeValidation.valid && (
                <p className="text-[11px] font-bold text-amber-500 text-center animate-pulse">ใส่ความยาวเรื่องให้ถูกก่อน เช่น 1.30 = 1 นาที 30 วินาที</p>
              )}
            </div>
            {inputFeedback && (
              <div className={`p-4 rounded-xl border flex items-start gap-3 animate-in slide-in-from-top-2 duration-300 relative ${inputFeedback.type === 'success' ? 'bg-[var(--ui-success)]/10 border-[var(--ui-success)]/20 text-[var(--ui-success)]' : inputFeedback.type === 'error' ? 'bg-[var(--ui-danger)]/10 border-[var(--ui-danger)]/20 text-[var(--ui-danger)]' : 'bg-[var(--ui-info)]/10 border-[var(--ui-info)]/20 text-[var(--ui-info)]'}`}>
                <span className="material-symbols-outlined text-lg mt-0.5">{inputFeedback.type === 'success' ? 'check_circle' : inputFeedback.type === 'error' ? 'error' : 'info'}</span>
                <div className="flex-1">
                  <p className="text-[12px] font-black uppercase tracking-tight">{inputFeedback.message}</p>
                  {inputFeedback.details && (inputFeedback.type === 'error' ? <ErrorDetails raw={inputFeedback.details} className="mt-0.5 opacity-80" /> : <p className="text-[10px] font-bold opacity-70 mt-0.5 whitespace-pre-wrap">{inputFeedback.details}</p>)}
                </div>
                <button onClick={() => setInputFeedback(null)} className="p-1 hover:bg-black/10 rounded transition-colors"><span className="material-symbols-outlined text-sm">close</span></button>
              </div>
            )}
          </div>
        </div>
        <div className="space-y-4 flex flex-col h-full">
          <div className="flex items-center justify-between">
            <div className="flex bg-[var(--ui-surface-soft)] p-1 rounded-xl border border-[var(--ui-border)] gap-1">
              <button onClick={() => setActiveTab('script')} disabled={loading} className={`px-4 py-1.5 rounded-lg text-[11px] font-extrabold uppercase tracking-widest transition-all ${activeTab === 'script' ? 'bg-[var(--ui-primary)] text-white shadow-md' : 'text-[var(--ui-text-muted)]'}`}>บทหนัง</button>
              <button onClick={() => setActiveTab('assets')} disabled={loading} className={`px-4 py-1.5 rounded-lg text-[11px] font-extrabold uppercase tracking-widest transition-all ${activeTab === 'assets' ? 'bg-[var(--ui-primary)] text-white shadow-md' : 'text-[var(--ui-text-muted)]'}`}>ตัวละคร & ฉาก</button>
            </div>
            <button onClick={() => setShowHistory(!showHistory)} disabled={loading} className="flex items-center gap-1 text-[var(--ui-accent)] hover:underline font-bold text-[11px] uppercase"><span className="material-symbols-outlined text-base">history</span> ประวัติ</button>
          </div>
          <div className="flex-1 relative min-h-[600px] bg-[var(--ui-input)] border border-[var(--ui-border)] rounded-2xl shadow-inner overflow-hidden flex flex-col">
            {!loading && activeTab === 'assets' && extractionAlert && (
              <div className="p-4 bg-amber-500/10 border-b border-amber-500/20 flex items-start gap-3 text-amber-500 animate-in slide-in-from-top-2">
                <span className="material-symbols-outlined text-lg mt-0.5">warning</span>
                <div className="flex-1"><p className="text-[12px] font-black uppercase tracking-tight">{extractionAlert.message}</p><p className="text-[10px] font-bold opacity-70 mt-0.5 whitespace-pre-wrap">{extractionAlert.details}</p></div>
                <button onClick={() => setExtractionAlert(null)} className="p-1 hover:bg-black/10 rounded transition-colors"><span className="material-symbols-outlined text-sm">close</span></button>
              </div>
            )}
            {loading && activeTask && (
              <div className="absolute inset-0 z-50 bg-[var(--ui-bg)]/80 backdrop-blur-sm flex flex-col items-center justify-center p-10 text-center animate-in fade-in duration-300">
                <div className="relative mb-6">
                  <div className={`w-20 h-20 border-4 rounded-full ${scriptMode === 'user' || activeTask.id === 'step1-assets' ? 'border-[var(--ui-accent)]/10' : 'border-[var(--ui-primary)]/10'}`} />
                  <div className={`absolute inset-0 w-20 h-20 border-4 border-t-transparent rounded-full animate-spin ${scriptMode === 'user' || activeTask.id === 'step1-assets' ? 'border-[var(--ui-accent)] shadow-[0_0_20px_rgba(245,158,11,0.2)]' : 'border-[var(--ui-primary)] shadow-lg'}`} />
                  <div className="absolute inset-0 flex items-center justify-center"><span className={`material-symbols-outlined text-3xl animate-pulse ${scriptMode === 'user' || activeTask.id === 'step1-assets' ? 'text-[var(--ui-accent)]' : 'text-[var(--ui-primary)]'}`}>{activeTask.id === 'step1-script' ? 'auto_awesome' : 'person_search'}</span></div>
                </div>
                <div className="space-y-3">
                  <h3 className="text-xl font-black text-[var(--ui-text-primary)] tracking-widest">{activeTask.progress}</h3>
                  <div className="flex items-center justify-center gap-3"><div className="px-3 py-1 bg-black/40 rounded-full border border-white/5"><span className={`font-mono font-black text-lg tabular-nums ${scriptMode === 'user' || activeTask.id === 'step1-assets' ? 'text-[var(--ui-accent)]' : 'text-[var(--ui-primary)]'}`}>{formatTimer(elapsed)}</span></div></div>
                </div>
                <button onClick={handleCancel} className="mt-10 px-8 py-3 bg-[var(--ui-surface-soft)] border border-[var(--ui-border)] text-[var(--ui-text-muted)] font-black rounded-xl hover:text-[var(--ui-text-primary)] hover:bg-[var(--ui-danger)]/10 hover:border-[var(--ui-danger)]/30 transition-all uppercase tracking-widest text-[11px]">ยกเลิก</button>
              </div>
            )}
            {activeTab === 'script' && (
              <div className="relative flex-1">
                {scriptMode === 'user' && processed && <div className="absolute top-4 left-6 z-10 px-3 py-1 bg-[var(--ui-accent)]/20 border border-[var(--ui-accent)]/30 rounded-full flex items-center gap-2"><span className="material-symbols-outlined text-[12px] text-[var(--ui-accent)] font-bold">lock</span><span className="text-[9px] font-black uppercase text-[var(--ui-accent)] tracking-widest">บทของคุณ — ระบบไม่แก้ไขเนื้อหา</span></div>}
                <textarea key={history[0]?.id || 'initial'} value={extractScreenplayOnly(processed)} onChange={(e) => onUpdate({ ...state.script, processed: extractAssetDefinitions(processed) ? `[ASSET DEFINITIONS]\n${extractAssetDefinitions(processed)}\n[END ASSET DEFINITIONS]\n\n${e.target.value}` : e.target.value })} className={`w-full h-full p-6 font-mono text-[13px] bg-transparent outline-none resize-none leading-relaxed text-[var(--ui-text-primary)] ${scriptMode === 'user' ? 'pt-14' : ''}`} placeholder="บทที่ประมวลผลเสร็จจะแสดงที่นี่..." />
              </div>
            )}
            {activeTab === 'assets' && (
              <div className={`w-full h-full p-6 overflow-y-auto ${!extractAssetDefinitions(processed) ? 'opacity-30' : ''}`}>
                {!extractAssetDefinitions(processed) ? <div className="flex flex-col items-center justify-center h-full text-center"><span className="material-symbols-outlined text-5xl mb-3">database_off</span><p className="text-sm font-bold uppercase tracking-widest">ยังไม่ได้สร้างองค์ประกอบ</p></div> : (
                  <div className="space-y-8">
                    <div className="flex justify-between items-center border-b border-[var(--ui-border)] pb-2 mb-2"><h4 className="text-[10px] font-black uppercase tracking-[0.2em] text-[var(--ui-accent)]">ข้อมูลองค์ประกอบ</h4><button onClick={handleReExtractAssets} disabled={loading || stopping} className="flex items-center gap-1.5 px-3 py-1 bg-[var(--ui-accent)]/10 text-[var(--ui-accent)] rounded-lg text-[10px] font-black uppercase tracking-widest hover:bg-[var(--ui-accent)]/20 transition-all border border-[var(--ui-accent)]/30 disabled:opacity-30">{stopping ? 'กำลังหยุดงานเดิม...' : <><span className="material-symbols-outlined text-sm font-bold">refresh</span>ดึงตัวละครและฉากใหม่</>}</button></div>
                    {Object.entries(assetData).map(([section, items]) => items.length > 0 && (
                        <div key={section} className="space-y-3">
                          <h4 className="text-[10px] font-black uppercase tracking-[0.2em] text-[var(--ui-text-muted)] pb-1">{section}</h4>
                          <table className="w-full text-left text-[11px] font-medium border-collapse">
                            <thead><tr className="text-[var(--ui-text-muted)] uppercase tracking-tighter"><th className="pb-2 pr-4 font-black">ID</th><th className="pb-2 pr-4 font-black">Name</th><th className="pb-2 font-black">Visual Prompt</th></tr></thead>
                            <tbody className="divide-y divide-[var(--ui-border)]/40">
                              {items.map((row, idx) => (<tr key={idx} className="group hover:bg-[var(--ui-surface-soft)]"><td className="py-2.5 pr-4 font-mono font-black text-[var(--ui-primary)]">{row[0]}</td><td className="py-2.5 pr-4 text-[var(--ui-text-primary)] font-bold">{row[1]}</td><td className="py-2.5 text-[var(--ui-text-muted)] italic leading-tight">{row[2]}</td></tr>))}
                            </tbody>
                          </table>
                        </div>
                    ))}
                  </div>
                )}
              </div>
            )}
            <div className="absolute bottom-4 right-6 bg-[var(--ui-bg)]/80 backdrop-blur px-3 py-1.5 rounded-full border border-[var(--ui-border)] shadow-sm"><span className="text-[10px] font-black uppercase text-[var(--ui-text-muted)]">ได้ <span className="text-[var(--ui-accent)]">{sceneCount}</span> ฉาก จากที่ต้องการ <span className="text-[var(--ui-text-primary)]">{metrics.editedShots}</span> ฉาก</span></div>
          </div>
        </div>
      </div>
      {showHistory && (
        <div className="fixed inset-0 z-[200] bg-black/90 backdrop-blur-md flex items-center justify-center p-6" onClick={() => setShowHistory(false)}>
          <div className="bg-[var(--ui-surface)] border border-[var(--ui-border)] w-full max-w-lg rounded-2xl overflow-hidden shadow-2xl flex flex-col" onClick={e => e.stopPropagation()}>
            <header className="p-5 border-b border-[var(--ui-border)] flex justify-between items-center bg-[var(--ui-surface-raised)]"><h3 className="text-sm font-black uppercase tracking-widest text-[var(--ui-text-primary)]">ประวัติเวอร์ชันบท (10 ล่าสุด)</h3><button onClick={() => setShowHistory(false)} className="material-symbols-outlined text-xl">close</button></header>
            <div className="flex-1 overflow-y-auto max-h-[400px] divide-y divide-[var(--ui-border)]">
              {history.map((item, i) => (
                <div key={item.id} className="p-5 hover:bg-[var(--ui-surface-soft)] transition-colors flex items-center justify-between group">
                  <div className="space-y-1"><p className="text-[11px] font-black text-[var(--ui-text-primary)] uppercase">{new Date(item.timestamp).toLocaleString('th-TH')}</p><p className="text-[10px] text-[var(--ui-text-muted)] font-bold uppercase tracking-tight">{item.sceneCount} ฉาก · {i === 0 ? 'เวอร์ชันปัจจุบัน' : `v.${history.length - i}`}</p></div>
                  <button onClick={() => { onUpdate({ ...state.script, processed: item.content }); setShowHistory(false); }} className="px-4 py-2 bg-[var(--ui-accent)] text-black rounded-lg text-[10px] font-black uppercase tracking-widest hover:bg-[var(--ui-accent-hover)] transition-all">ใช้เวอร์ชันนี้</button>
                </div>
              ))}
              {history.length === 0 && <div className="p-10 text-center text-[var(--ui-text-muted)] font-bold text-xs uppercase">ยังไม่มีประวัติการบันทึก</div>}
            </div>
          </div>
        </div>
      )}
      <ConfirmDialog
        open={showOverwriteConfirm}
        icon="warning"
        tone="warning"
        title="มีบทเดิมอยู่แล้ว"
        message="ต้องการสร้างบทใหม่ทับหรือไม่ — บทเดิมเรียกคืนได้จากปุ่ม ประวัติ"
        confirmLabel="สร้างบทใหม่ทับ"
        onConfirm={executeGeneration}
        onCancel={() => setShowOverwriteConfirm(false)}
      />
      <div className="fixed bottom-0 left-0 lg:left-[280px] right-0 bg-[var(--ui-bg)]/90 backdrop-blur-2xl border-t border-[var(--ui-border)] p-6 flex justify-end items-center z-40"><NextStepButton onNext={onNext} missingRequirements={missingRequirements} className="px-12 py-3.5 bg-[var(--ui-accent)] text-black font-extrabold rounded-full shadow-2xl uppercase tracking-widest" /></div>
    </div>
  );
};
