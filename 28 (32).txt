import { Flow } from 'flow-sdk';
import { parseAssetLines, AssetLine } from './assetLineParser';
import { AssetType } from '../types';
const TASKS: Record<AssetType, string> = {
  Character: `list up to 12 people who appear, speak or are clearly seen, including minor characters with lines. Output one per line: Name exactly as written in the script (same language and alphabet, never romanized) | English visual description starting with ethnicity, then age, gender, build, face, hair and outfit.`,
  Location: `list up to 8 distinct physical places where scenes happen. Parts of the same place are ONE location: a counter, kitchen, doorway or corner inside the same building belongs to that building. Time of day, weather or lighting never creates a new location. Output one per line: Short place name in the script's own language and alphabet | English visual description of architecture, materials, layout and key objects.`,
  Prop: `list up to 12 physical objects that a character holds, uses, gives, receives or reacts to, or that the plot depends on. Skip furniture, buildings and background items. Output one item per line: Name exactly as written in the script (same language and alphabet, never romanized) | English visual description of material, shape, color and size.`
};
export async function pullAssetItems(
  type: AssetType,
  scriptBody: string,
  worldEn: string,
  thinkingLevel: 'high' | 'medium' | 'low',
  onProgress?: (text: string) => void,
  isCancelled?: () => boolean
): Promise<{ items: AssetLine[]; failedParts: string[]; okCount: number; lastError: any }> {
  const chunks: string[] = [];
  let buf = '';
  for (const line of scriptBody.split('\n')) {
    if (buf.length + line.length > 2500 && buf.trim()) { chunks.push(buf.trim()); buf = ''; }
    buf += line + '\n';
  }
  if (buf.trim()) chunks.push(buf.trim());
  const ask = async (text: string): Promise<AssetLine[]> => {
    const prompt = `Story world: ${worldEn}. From the script below, ${TASKS[type]} No numbering, no other text.\n\nSCRIPT:\n${text}`;
    const timeout = new Promise((_, reject) => setTimeout(() => reject(new Error('TIMEOUT')), 120000));
    const res: any = await Promise.race([
      Flow.generate.text(prompt, { systemInstruction: 'You are a professional production designer. Return ONLY lines of text.', thinkingLevel }),
      timeout
    ]);
    const out = typeof res?.text === 'string' ? res.text : '';
    if (!out.trim()) throw new Error('AI ไม่ได้ส่งข้อความกลับ');
    return parseAssetLines(out);
  };
  const items: AssetLine[] = [];
  const seen = new Set<string>();
  const add = (list: AssetLine[]) => list.forEach(it => {
    const key = it.name.trim().toLowerCase();
    if (it.name && it.desc && !seen.has(key)) { seen.add(key); items.push(it); }
  });
  const failedParts: string[] = [];
  let okCount = 0;
  let lastError: any = null;
  for (let i = 0; i < chunks.length; i++) {
    if (isCancelled?.()) throw new Error('CANCELLED');
    onProgress?.(`ช่วง ${i + 1}/${chunks.length}`);
    try {
      add(await ask(chunks[i]));
      okCount++;
    } catch (err) {
      lastError = err;
      const lines = chunks[i].split('\n');
      const half = Math.ceil(lines.length / 2);
      const parts = [lines.slice(0, half).join('\n'), lines.slice(half).join('\n')].filter(p => p.trim());
      for (let k = 0; k < parts.length; k++) {
        if (isCancelled?.()) throw new Error('CANCELLED');
        onProgress?.(`ช่วง ${i + 1}/${chunks.length} (แบ่งย่อย ${k + 1}/${parts.length})`);
        try { add(await ask(parts[k])); okCount++; }
        catch (err2) { lastError = err2; failedParts.push(parts[k].replace(/\s+/g, ' ').slice(0, 60)); }
      }
    }
  }
  return { items, failedParts, okCount, lastError };
}
