function toPositions(v: any): string | undefined {
  if (Array.isArray(v)) return v.map(x => typeof x === 'string' ? x : Object.values(x || {}).join(' ')).join(', ').trim() || undefined;
  if (v && typeof v === 'object') return Object.entries(v).map(([k, x]) => `${k} ${x}`).join(', ').trim() || undefined;
  return typeof v === 'string' && v.trim() ? v.trim().slice(0, 200) : undefined;
}

function getSimilarity(s1: string, s2: string): number {
  if (!s1 || !s2) return 0;
  const clean1 = s1.toLowerCase().replace(/\s+/g, '');
  const clean2 = s2.toLowerCase().replace(/\s+/g, '');
  if (clean1 === clean2) return 1;
  const getBigrams = (str: string) => {
    const set = new Set<string>();
    for (let i = 0; i < str.length - 1; i++) {
      set.add(str.slice(i, i + 2));
    }
    return set;
  };
  const b1 = getBigrams(clean1);
  const b2 = getBigrams(clean2);
  const intersection = new Set([...b1].filter(x => b2.has(x)));
  return (2 * intersection.size) / ((b1.size + b2.size) || 1);
}
function repairTruncatedJson(text: string): any[] | null {
  if (!text) return null;
  const clean = text.replace(/```json/g, '').replace(/```/g, '').trim();
  const lastBrace = clean.lastIndexOf('}');
  if (lastBrace === -1) return null;
  let partial = clean.substring(0, lastBrace + 1).trim();
  const firstBracket = partial.indexOf('[');
  if (firstBracket === -1) return null;
  partial = partial.substring(firstBracket);
  if (!partial.endsWith(']')) partial += ']';
  try {
    const parsed = JSON.parse(partial);
    return Array.isArray(parsed) ? parsed : null;
  } catch (e) {
    return null;
  }
}

                              {(s.videoModel || s.durationManual) && <button onClick={() => setShotClip(s.id, { videoModel: undefined, durationManual: false })} className="flex-1 py-2.5 rounded-xl border border-[var(--ui-border)] text-[12px] font-bold text-[var(--ui-text-primary)] hover:bg-[var(--ui-surface-soft)]">ใช้ค่าทั้งเรื่อง</button>}
                              <button onClick={() => setActiveAssetEdit(null)} className="flex-1 py-2.5 rounded-xl bg-[var(--ui-accent)] text-black text-[12px] font-extrabold">เสร็จ</button>
                            </div>
                          </div>
                        </div>
                      )}
                    </div>
                  </div>
                </div>
              </div>
            );
          })
        )}
      </div>
      <Pager page={currentPage} total={state.storyboard.length} perPage={ITEMS_PER_PAGE} onChange={goToPage} />
      <ConfirmDialog
        open={!!showMigrationConfirm}
        icon="photo_library"
        title="ตรวจพบภาพเดิม"
        message={<>มีภาพเดิม <span className="text-[var(--ui-accent)] font-extrabold">{showMigrationConfirm?.count || 0} ช็อต</span> ต้องการย้ายภาพไปช็อตใหม่ที่ใกล้เคียงกันไหม</>}
        confirmLabel="ย้ายภาพ (แนะนำ)"
        onConfirm={() => { if (showMigrationConfirm) finalizeMigration(showMigrationConfirm.newShots, showMigrationConfirm.failures || []); }}
        extraActions={[{
          label: 'เริ่มใหม่ ไม่ใช้ภาพเดิม',
          onClick: () => { 
            if (showMigrationConfirm) {
              onUpdate({ storyboard: showMigrationConfirm.newShots, splitFailures: showMigrationConfirm.failures || [] }); 
              setCurrentPage(1);
              setShowMigrationConfirm(null); 
              clearTask('split-shots'); 
            }
          }
        }]}
        cancelLabel="ยกเลิก (เก็บช็อตเดิม)"
        onCancel={() => { setShowMigrationConfirm(null); clearTask('split-shots'); }}
      />
      <ConfirmDialog
        open={confirmQueue}
        icon="auto_awesome_motion"
        tone="warning"
        title="สร้างภาพทุกช็อต"
        message={<>จะสร้างภาพ <span className="text-[var(--ui-accent)] font-extrabold">{queueStats.targetCount} ช็อต</span>{queueStats.oldCount > 0 ? ` (รวมภาพเก่า ${queueStats.oldCount} ช็อต — ภาพเดิมจะถูกแทนที่)` : ''}</>}
        confirmLabel="เริ่มสร้าง"
        confirmDisabled={queueStats.targetCount === 0}
        onConfirm={handleQueueAll}
        onCancel={() => setConfirmQueue(false)}
      />
      <ConfirmDialog
        open={confirmApplyAll}
        icon="restart_alt"
        tone="warning"
        title="ใช้ค่าทั้งเรื่องกับทุกช็อต"
        message="ล้างโมเดลและความยาวที่ตั้งเองรายช็อต แล้วใช้ค่าทั้งเรื่องแทน"
        confirmLabel="ใช้กับทุกช็อต"
        onConfirm={applyAllShots}
        onCancel={() => setConfirmApplyAll(false)}
      />
      {uploadError && (
        <div className="fixed bottom-28 left-1/2 -translate-x-1/2 z-[60] flex items-center gap-3 px-5 py-3 bg-[var(--ui-danger)] text-white rounded-xl shadow-2xl">
          <span className="material-symbols-outlined font-bold">error</span>
          <span className="font-bold text-sm">{uploadError}</span>
          <button onClick={() => setUploadError(null)} className="material-symbols-outlined">close</button>
        </div>
      )}
      <div className="fixed bottom-0 left-0 lg:left-[280px] right-0 bg-[var(--ui-bg)]/90 backdrop-blur-2xl border-t border-[var(--ui-border)] p-6 flex justify-end items-center z-40"><NextStepButton onNext={onNext} missingRequirements={missingRequirements} className="px-12 py-3.5 bg-amber-500 text-black font-extrabold rounded-full shadow-2xl uppercase tracking-widest" /></div>
    </div>
  );
};
