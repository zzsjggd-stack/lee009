import React, { useState, useRef, useEffect } from 'react';
import { Flow } from 'flow-sdk';
import { Asset, AppState } from '../types';
import { STYLES_PROMPT } from '../lib/constants';
import { resolveStoryWorld } from '../lib/storyWorld';
import { isAssetImageOld } from '../lib/staleness';
import { identityRequestPrompt } from '../lib/castPrompt';
interface Props {
  asset: Asset;
  state: AppState;
  onClose: () => void;
  onUpdate: (patch: Partial<Asset>) => void;
  onUploadReference: () => void;
  onUploadFinal: (file?: File) => void;
  onGenerate: (asset: Asset) => void;
}
const AI_TIMEOUT = 60000; // 60 seconds for translation/prompt tasks
export const AssetEditPanel: React.FC<Props> = ({ asset, state, onClose, onUpdate, onUploadFinal, onGenerate }) => {
  const [isExpanding, setIsExpanding] = useState(false);
  const [isTranslating, setIsTranslating] = useState(false);
  const [isConverting, setIsConverting] = useState(false);
  const [isWritingIdentity, setIsWritingIdentity] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);
  const [hasEnglishChanged, setHasEnglishChanged] = useState(false);
  
  // Multiple Upload State
  const [isUploading, setIsUploading] = useState(false);
  const [uploadError, setUploadError] = useState<string | null>(null);
  
  const fileInputRef = useRef<HTMLInputElement>(null);
  const multiUploadRef = useRef<HTMLInputElement>(null);
  const lastSyncRef = useRef(asset.visualPrompt);
  // Initial translation if Thai is missing
  useEffect(() => {
    if (asset.visualPrompt && !asset.visualPromptThai) {
      translateEnglishToThai();
    }
  }, []);
  // Track if English prompt has been edited manually
  useEffect(() => {
    if (asset.visualPrompt !== lastSyncRef.current) {
      setHasEnglishChanged(true);
    }
  }, [asset.visualPrompt]);
  const generateWithTimeout = async (prompt: string, instruction: string) => {
    setErrorMessage(null);
    const timeoutPromise = new Promise((_, reject) => 
      setTimeout(() => reject(new Error('TIMEOUT')), AI_TIMEOUT)
    );
    
    try {
      const result = await Promise.race([
        Flow.generate.text(prompt, {
          systemInstruction: instruction,
          thinkingLevel: state.config.textModel 
        }),
        timeoutPromise
      ]) as any;
      return result.text.trim();
    } catch (e: any) {
      if (e.message === 'TIMEOUT') {
        setErrorMessage('AI ไม่ตอบกลับภายใน 60 วินาที กรุณาลองใหม่');
      } else {
        setErrorMessage('การเรียก AI ล้มเหลว: ' + e.message);
      }
      throw e;
    }
  };
  const translateEnglishToThai = async () => {
    if (!asset.visualPrompt) return;
    setIsTranslating(true);
    try {
      const prompt = `Translate the following English image prompt into simple, natural, and easy-to-read Thai for a creative person. Focus on clear visual descriptions. English prompt: ${asset.visualPrompt}`;
      const instruction = "You are a creative translator. Output only the natural Thai translation, no explanations.";
      const thaiText = await generateWithTimeout(prompt, instruction);
      // Translation from English does not count as manual user edit
      onUpdate({ visualPromptThai: thaiText });
      lastSyncRef.current = asset.visualPrompt;
      setHasEnglishChanged(false);
    } catch (e) {
      console.error(e);
    } finally {
      setIsTranslating(false);
    }
  };
  const convertThaiToEnglishPrompt = async () => {
    if (!asset.visualPromptThai) return;
    setIsConverting(true);
    try {
      const world = resolveStoryWorld(state.script);
      // Find ethnicity prefix if exists in context or original prompt
      const ethnicityMatch = asset.visualPrompt?.match(/^(Thai|Chinese|Japanese|Korean|Caucasian|Asian|Person)\s(man|woman|person|child)/i);
      const prefix = ethnicityMatch ? ethnicityMatch[0] : "";
      const prompt = `Translate the following Thai visual description into a professional English image prompt keywords. 
      Rules: 
      1. Use single line, comma-separated keywords.
      2. Must start with "${prefix || 'A person'}" if it's a character.
      3. Omit any words about artistic style, background scenery, lighting, or camera technicals (they are added later).
      4. Story world: ${world.en}. Keep the era and clothing exactly as the details say; if they do not say, make it fit this story world.
      Thai text: ${asset.visualPromptThai}`;
      
      const instruction = "You are an expert prompt engineer. Output only the comma-separated English prompt keywords.";
      const englishPrompt = await generateWithTimeout(prompt, instruction);
      
      // Requirement 2: Manual conversion from AI tools sets promptEditedByUser
      onUpdate({ visualPrompt: englishPrompt, promptEditedByUser: true });
      lastSyncRef.current = englishPrompt;
      setHasEnglishChanged(false);
    } catch (e) {
      console.error(e);
    } finally {
      setIsConverting(false);
    }
  };
  const handleExpandPrompt = async () => {
    setIsExpanding(true);
    try {
      const world = resolveStoryWorld(state.script);
      const globalStyle = STYLES_PROMPT[state.config.style] || state.config.style;
      const prompt = `
        Write a detailed visual concept description for this ${asset.type}.
        Name: ${asset.name}
        Details: ${asset.description}
        Overall Style: ${globalStyle}
        Metadata: ${JSON.stringify(asset.metadata)}
        
        Rules:
        - Output a single descriptive paragraph in English.
        - Focus on visual traits matching the Overall Style.
        - Story world: ${world.en}. Keep the era and clothing exactly as the details say; if they do not say, make it fit this story world.
        ${asset.type === 'Character' ? '- Always describe the outfit.' : ''}
      `;
      const res = await generateWithTimeout(prompt, "You are a professional production designer. Output a dense visual prompt.");
      // Requirement 2: Manual conversion from AI tools sets promptEditedByUser
      onUpdate({ visualPrompt: res, promptEditedByUser: true });
      lastSyncRef.current = res;
      setHasEnglishChanged(false);
    } catch (e) {
      console.error(e);
    } finally {
      setIsExpanding(false);
    }
  };
  const handleWriteIdentity = async () => {
    setIsWritingIdentity(true);
    try {
      const res = await generateWithTimeout(
        identityRequestPrompt(asset),
        "You are a casting director. Output one short line only."
      );
      if (asset.type === 'Character') {
        onUpdate({ metadata: { ...asset.metadata, identityLock: res } });
      } else {
        onUpdate({ metadata: { ...asset.metadata, spatialLock: res } });
      }
    } catch (e) {
      console.error(e);
    } finally {
      setIsWritingIdentity(false);
    }
  };
  const handleMultiUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const files = e.target.files;
    if (!files || files.length === 0) return;
    setUploadError(null);
    const existingCount = asset.sourceReferences?.length || 0;
    
    if (existingCount + files.length > 3) {
      setUploadError('เพิ่มรูปไม่ได้: รวมสูงสุดต้องไม่เกิน 3 รูป');
      return;
    }
    setIsUploading(true);
    const newRefs = [...(asset.sourceReferences || [])];
    try {
      for (let i = 0; i < files.length; i++) {
        const file = files[i];
        
        // Validation
        const validTypes = ['image/png', 'image/jpeg', 'image/webp'];
        if (!validTypes.includes(file.type)) {
          throw new Error(`ไฟล์ ${file.name} ไม่ใช่ PNG, JPG หรือ WEBP`);
        }
        if (file.size > 10 * 1024 * 1024) {
          throw new Error(`ไฟล์ ${file.name} มีขนาดเกิน 10MB`);
        }
        const base64 = await new Promise<string>((resolve, reject) => {
          const reader = new FileReader();
          reader.onload = () => resolve((reader.result as string).split(',')[1]);
          reader.onerror = reject;
          reader.readAsDataURL(file);
        });
        const uploaded = await Flow.upload({ 
          base64, 
          mimeType: file.type as any, 
          name: `Ref: ${asset.name}` 
        });
        newRefs.push({ mediaId: uploaded.mediaId, base64 });
      }
      onUpdate({ sourceReferences: newRefs });
    } catch (err: any) {
      setUploadError(err.message || 'อัปโหลดล้มเหลว');
    } finally {
      setIsUploading(false);
      if (multiUploadRef.current) multiUploadRef.current.value = '';
    }
  };
  const isImageOld = isAssetImageOld(asset, state.config);
  return (
    <div className="fixed inset-0 z-[100] flex items-center justify-end bg-black/80 backdrop-blur-sm p-4 md:p-6">
      <div className="bg-[var(--ui-surface)] border border-[var(--ui-border-strong)] w-full max-w-2xl h-full rounded-3xl shadow-2xl flex flex-col overflow-hidden animate-in slide-in-from-right-10 duration-300">
        <input 
          ref={fileInputRef} 
          type="file" 
          accept="image/*" 
          className="hidden" 
          onChange={(e) => { const file = e.target.files?.[0]; e.target.value = ''; if (file) onUploadFinal(file); }} 
        />
        <input ref={multiUploadRef} type="file" multiple accept=".png,.jpg,.jpeg,.webp" className="hidden" onChange={handleMultiUpload} />
        <header className="p-6 border-b border-[var(--ui-border)] flex items-center justify-between bg-[var(--ui-surface-raised)]">
          <div className="flex items-center gap-4">
             <div className="w-12 h-12 bg-[var(--ui-accent)]/10 rounded-xl flex items-center justify-center text-[var(--ui-accent)] shadow-inner">
               <span className="material-symbols-outlined text-2xl font-bold">
                 {asset.type === 'Character' ? 'person' : asset.type === 'Location' ? 'distance' : 'category'}
               </span>
             </div>
             <div>
               <h3 style={{ fontSize: 'var(--fs-h2)' }} className="font-extrabold text-[var(--ui-text-primary)] tracking-tight">{asset.name}</h3>
               <p style={{ fontSize: '10px' }} className="font-mono text-[var(--ui-text-muted)] font-bold uppercase tracking-widest">{asset.id} • {asset.type.toUpperCase()}</p>
             </div>
          </div>
          <button onClick={onClose} className="p-2 text-[var(--ui-text-muted)] hover:text-[var(--ui-text-primary)] transition-colors active:scale-90">
            <span className="material-symbols-outlined text-2xl font-bold">close</span>
          </button>
        </header>
        <div className="flex-1 overflow-y-auto p-8 md:p-9 space-y-8 custom-scrollbar">
          
          <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
            <div className="space-y-2">
              <label style={{ fontSize: '9px' }} className="font-extrabold uppercase tracking-[0.15em] text-[var(--ui-text-muted)] pl-0.5">ชื่อ (NAME)</label>
              <input 
                type="text" 
                value={asset.name} 
                onChange={e => onUpdate({ name: e.target.value, nameEditedByUser: true })} 
                className="w-full bg-[var(--ui-input)] border border-[var(--ui-border)] rounded-xl px-4 h-11 text-[var(--fs-small)] text-[var(--ui-text-primary)] outline-none focus:border-[var(--ui-accent)]/60 shadow-inner font-medium transition-colors"
              />
            </div>
            <div className="space-y-2">
              <label style={{ fontSize: '9px' }} className="font-extrabold uppercase tracking-[0.15em] text-[var(--ui-text-muted)] pl-0.5">ชื่อเล่น / ชื่ออื่น (NICKNAMES)</label>
              <input 
                type="text" 
                value={asset.aliases || ''} 
                onChange={e => onUpdate({ aliases: e.target.value })} 
                placeholder="เช่น พี่เข้ม, ไอ้เข้ม, ท่านเข้ม"
                className="w-full bg-[var(--ui-input)] border border-[var(--ui-border)] rounded-xl px-4 h-11 text-[var(--fs-small)] text-[var(--ui-text-primary)] outline-none focus:border-[var(--ui-accent)]/60 shadow-inner font-medium transition-colors"
              />
              <p style={{ fontSize: '9px' }} className="text-[var(--ui-text-muted)] font-bold pl-1">ใช้ให้ระบบจับคู่ตัวละครในบทได้ถูก</p>
            </div>
          </div>
          {asset.type === 'Character' && (
            <div className="space-y-2">
              <label style={{ fontSize: '9px' }} className="font-extrabold uppercase tracking-[0.15em] text-[var(--ui-text-muted)] pl-0.5">เสียงพูด (VOICE) — ภาษาอังกฤษ</label>
              <textarea
                value={asset.voiceProfile || ''}
                onChange={(e) => onUpdate({ voiceProfile: e.target.value })}
                placeholder="deep raspy male voice, around 45 years old, slow and calm, slight Thai northern accent"
                className="w-full h-20 bg-[var(--ui-input)] border border-[var(--ui-border)] rounded-xl p-4 text-[11px] font-mono text-[var(--ui-text-primary)] outline-none focus:border-[var(--ui-accent)]/60 resize-none leading-relaxed shadow-inner"
              />
              <p style={{ fontSize: '9px' }} className="text-[var(--ui-text-muted)] font-bold pl-1">ระบบจะส่งข้อความนี้เข้าทุกช็อตที่ตัวละครนี้พูด เพื่อให้เสียงเหมือนเดิมทั้งเรื่อง</p>
            </div>
          )}
          {(asset.type === 'Character' || asset.type === 'Location') && (
            <div className="space-y-2">
              <div className="flex justify-between items-center">
                <label style={{ fontSize: '9px' }} className="font-extrabold uppercase tracking-[0.15em] text-[var(--ui-text-muted)] pl-0.5">
                  {asset.type === 'Character' ? 'ลักษณะจำเพาะ (ล็อกหน้า) — ภาษาอังกฤษ' : 'ลักษณะจำเพาะสถานที่ (ล็อกฉาก) — ภาษาอังกฤษ'}
                </label>
                <button 
                  onClick={handleWriteIdentity}
                  disabled={isWritingIdentity}
                  className="flex items-center gap-1 text-[10px] font-bold text-[var(--ui-accent)] hover:text-[var(--ui-accent-hover)] transition-colors uppercase tracking-widest disabled:opacity-30"
                >
                  {isWritingIdentity ? <span className="material-symbols-outlined text-sm animate-spin">sync</span> : <span className="material-symbols-outlined text-sm">auto_awesome</span>}
                  ให้ AI เขียนให้
                </button>
              </div>
              <textarea
                value={(asset.type === 'Character' ? asset.metadata.identityLock : asset.metadata.spatialLock) || ''}
                onChange={(e) => {
                  const val = e.target.value;
                  if (asset.type === 'Character') onUpdate({ metadata: { ...asset.metadata, identityLock: val } });
                  else onUpdate({ metadata: { ...asset.metadata, spatialLock: val } });
                }}
                placeholder={asset.type === 'Character' 
                  ? "woman about 25, oval face, mole under right eye, long straight black hair, fair pink skin, slim" 
                  : "wooden Thai house, teak floor, red lacquer cabinet on the left wall, round window at the back"}
                className="w-full h-20 bg-[var(--ui-input)] border border-[var(--ui-border)] rounded-xl p-4 text-[11px] font-mono text-[var(--ui-text-primary)] outline-none focus:border-[var(--ui-accent)]/60 resize-none leading-relaxed shadow-inner"
              />
              <p style={{ fontSize: '9px' }} className="text-[var(--ui-text-muted)] font-bold pl-1">
                ส่งเข้าทุกช็อตเพื่อล็อกหน้า/ฉาก — เขียนสั้นๆ เฉพาะจุดที่ทำให้ต่างจากตัวอื่น ถ้าเว้นว่าง ระบบใช้ 25 คำแรกของพรอมต์แทน
              </p>
            </div>
          )}
          <div className="space-y-3">
             <div className="flex justify-between items-end px-1">
                <label style={{ fontSize: 'var(--fs-label)' }} className="font-bold uppercase tracking-widest text-[var(--ui-accent)] tracking-thai-tight">คำบรรยาย (ภาษาไทย)</label>
                {hasEnglishChanged && (
                  <button 
                    onClick={translateEnglishToThai}
                    disabled={isTranslating}
                    className="flex items-center gap-1 text-[10px] font-bold text-[var(--ui-accent)] hover:text-[var(--ui-accent-hover)] transition-colors uppercase tracking-widest disabled:opacity-30"
                  >
                    <span className="material-symbols-outlined text-sm">translate</span>
                    แปลเป็นไทยใหม่
                  </button>
                )}
             </div>
             <div className="relative">
                <textarea 
                  value={asset.visualPromptThai || ''}
                  onChange={(e) => onUpdate({ visualPromptThai: e.target.value, promptEditedByUser: true })}
                  placeholder="เขียนคำอธิบายภาพเป็นภาษาไทยที่นี่..."
                  className={`w-full h-32 bg-[var(--ui-input)] border border-[var(--ui-border)] rounded-xl p-4 text-[var(--fs-small)] text-[var(--ui-text-primary)] outline-none focus:border-[var(--ui-accent)]/60 resize-none leading-relaxed shadow-inner transition-opacity ${isTranslating ? 'opacity-30' : ''}`}
                />
                {isTranslating && (
                  <div className="absolute inset-0 flex items-center justify-center">
                    <div className="w-8 h-8 border-2 border-[var(--ui-accent)] border-t-transparent rounded-full animate-spin" />
                  </div>
                )}
             </div>
             <button 
                onClick={convertThaiToEnglishPrompt}
                disabled={isConverting || !asset.visualPromptThai}
                className="w-full py-2 bg-[var(--ui-surface-soft)] border border-[var(--ui-border)] rounded-lg text-[10px] font-bold uppercase tracking-widest text-[var(--ui-accent)] hover:bg-[var(--ui-surface-raised)] transition-all flex items-center justify-center gap-2 disabled:opacity-30"
             >
                {isConverting ? <div className="w-3 h-3 border-2 border-current border-t-transparent rounded-full animate-spin" /> : <span className="material-symbols-outlined text-sm">swap_horiz</span>}
                แปลงเป็นพรอมต์
             </button>
          </div>
          <div className="space-y-3">
             <div className="flex justify-between items-center px-1">
                <label style={{ fontSize: '11px' }} className="font-bold uppercase tracking-widest text-emerald-500 tracking-thai-tight">พรอมต์ที่ส่งจริง (อังกฤษ)</label>
                <button 
                  onClick={handleExpandPrompt}
                  disabled={isExpanding}
                  className="flex items-center gap-1 text-[10px] font-bold text-emerald-500 hover:text-emerald-400 transition-colors uppercase tracking-widest disabled:opacity-30"
                >
                  <span className="material-symbols-outlined text-sm">auto_awesome</span>
                  เรียบเรียงใหม่
                </button>
             </div>
             <div className="relative">
                <textarea 
                  value={asset.visualPrompt || ''}
                  onChange={(e) => onUpdate({ visualPrompt: e.target.value, promptEditedByUser: true })}
                  style={{ fontSize: '11px', height: '100px' }}
                  className={`w-full bg-[var(--ui-input)] border border-[var(--ui-border)] rounded-xl p-4 font-mono text-emerald-400/90 outline-none focus:border-emerald-600/60 resize-none leading-relaxed shadow-inner ${isExpanding ? 'opacity-30' : ''}`}
                />
                {isExpanding && (
                  <div className="absolute inset-0 flex items-center justify-center">
                    <div className="w-6 h-6 border-2 border-emerald-500 border-t-transparent rounded-full animate-spin" />
                  </div>
                )}
             </div>
             {errorMessage && <p className="text-[10px] font-bold text-red-500 px-1">{errorMessage}</p>}
             <p className="text-[10px] text-[var(--ui-text-muted)] font-bold px-1 opacity-60">ช่องนี้คือสิ่งที่ใช้สร้างภาพจริง</p>
          </div>
          <div className="space-y-5">
              <div className="px-1">
                <label style={{ fontSize: 'var(--fs-label)' }} className="font-bold uppercase tracking-widest text-[var(--ui-text-muted)] tracking-thai-tight block">รูปอ้างอิง (REFERENCE CLOSET)</label>
                <p className="text-[10px] text-[var(--ui-text-muted)] mt-0.5 font-medium">ใส่รูปหน้าคน ชุด หรือสิ่งของจริง ให้ AI สร้างชีทตรงขึ้น (สูงสุด 3 รูป)</p>
              </div>
              <div className="grid grid-cols-4 gap-4">
                  {asset.sourceReferences?.map((ref, idx) => (
                    <div key={idx} className="aspect-square rounded-xl overflow-hidden border border-[var(--ui-border)] relative group shadow-lg">
                      {ref.base64 ? (
                        <img src={`data:image/png;base64,${ref.base64}`} className="w-full h-full object-cover" alt={`Ref ${idx}`} />
                      ) : (
                        <div className="w-full h-full flex flex-col items-center justify-center bg-[var(--ui-surface-soft)] text-center p-2">
                           <span className="material-symbols-outlined text-[var(--ui-accent)] text-lg mb-1">cloud_done</span>
                           <p className="text-[8px] text-[var(--ui-text-muted)] font-bold leading-tight">อยู่ในคลัง Flow</p>
                        </div>
                      )}
                      <button 
                        onClick={() => onUpdate({ sourceReferences: asset.sourceReferences?.filter((_, i) => i !== idx) })}
                        className="absolute inset-0 bg-red-600/80 opacity-0 group-hover:opacity-100 flex items-center justify-center transition-all"
                      >
                        <span className="material-symbols-outlined text-white text-xl font-bold">delete</span>
                      </button>
                    </div>
                  ))}
                  {isUploading && (
                    <div className="aspect-square border-2 border-dashed border-[var(--ui-accent)]/40 bg-[var(--ui-accent)]/5 rounded-xl flex items-center justify-center">
                       <div className="w-6 h-6 border-2 border-[var(--ui-accent)] border-t-transparent rounded-full animate-spin" />
                    </div>
                  )}
                  {(asset.sourceReferences?.length || 0) < 3 && !isUploading && (
                    <button 
                      onClick={() => multiUploadRef.current?.click()}
                      className="aspect-square border-2 border-dashed border-[var(--ui-border)] rounded-xl flex flex-col items-center justify-center text-[var(--ui-text-muted)] hover:text-[var(--ui-text-secondary)] hover:border-[var(--ui-text-muted)] transition-all gap-1.5 active:scale-95"
                    >
                      <span className="material-symbols-outlined text-2xl font-bold">add_photo_alternate</span>
                      <span style={{ fontSize: '9px' }} className="font-black uppercase tracking-widest">เพิ่มรูป</span>
                    </button>
                  )}
              </div>
              {uploadError && <p className="text-[10px] font-bold text-red-500 px-1">{uploadError}</p>}
          </div>
          <div className="pt-2">
              <button 
                onClick={() => fileInputRef.current?.click()}
                className="w-full py-7 border-2 border-dashed border-emerald-500/20 bg-emerald-500/5 text-emerald-500 rounded-2xl font-extrabold uppercase flex flex-col items-center gap-2 hover:bg-emerald-500/10 transition-all active:scale-[0.99]"
              >
                <span className="material-symbols-outlined text-3xl font-bold">upload_file</span>
                <span style={{ fontSize: 'var(--fs-body)' }}>แทนที่ภาพด้วยไฟล์ของฉัน</span>
                <span style={{ fontSize: '9px' }} className="font-bold opacity-60 uppercase tracking-widest mt-0.5">ใช้รูปนี้เป็น Master ทันที</span>
              </button>
          </div>
        </div>
        <footer className="p-8 border-t border-[var(--ui-border)] bg-[var(--ui-surface-raised)] flex flex-col gap-4">
           {isImageOld && (
             <div className="bg-amber-500 text-black px-4 py-2 rounded-xl text-[11px] font-black uppercase text-center animate-pulse">
               ภาพยังเป็นแบบเก่า — กดสร้างใหม่
             </div>
           )}
           <div className="flex gap-4">
             <button onClick={onClose} className="flex-1 py-4 font-bold uppercase tracking-widest text-[var(--ui-text-muted)] hover:text-[var(--ui-text-primary)] transition-colors text-sm">
               ปิดหน้าต่าง
             </button>
             <button onClick={() => onGenerate(asset)} className="flex-[2] py-4 g2-cta rounded-2xl flex items-center justify-center gap-2 shadow-xl transition-all active:scale-95 text-sm">
