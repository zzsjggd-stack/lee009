  people: string;  // คำระบุเชื้อชาติ/กลุ่มคนหลัก (อังกฤษ)
  outfit: string;  // คำระบุสไตล์การแต่งกายพื้นฐาน (อังกฤษ)
  th: string;      // ชื่อเรียกในแอป (ภาษาไทย)
}
/**
 * แปลงค่า "โลกของเรื่อง" จากบทหนังให้เป็นข้อมูลเมทาดาตาที่ทุกขั้นตอนใช้งานตรงกัน
 */
export function resolveStoryWorld(script: ScriptData): StoryWorld {
  const { setting, customSetting, customSettingThai } = script;
  switch (setting) {
    case 'ไทยร่วมสมัย':
      return {
        en: 'Contemporary Thailand, present day',
        people: 'Thai',
        outfit: 'modern everyday clothing',
        th: 'ไทยร่วมสมัย'
      };
    case 'ไทยย้อนยุค':
      return {
        en: 'Historical Thailand (old Siam), the exact period follows the script',
        people: 'Thai',
        outfit: 'traditional Siamese period costume',
        th: 'ไทยย้อนยุค'
      };
    case 'จีนร่วมสมัย':
      return {
        en: 'Contemporary China, present day',
        people: 'Chinese',
        outfit: 'modern everyday clothing',
        th: 'จีนร่วมสมัย'
      };
    case 'จีนย้อนยุค':
      return {
        en: 'Ancient imperial China, the exact dynasty follows the script',
        people: 'Chinese',
        outfit: 'traditional Chinese period costume (hanfu)',
        th: 'จีนย้อนยุค'
      };
    case 'เกาหลีร่วมสมัย':
      return {
        en: 'Contemporary South Korea, present day',
        people: 'Korean',
        outfit: 'modern everyday clothing',
        th: 'เกาหลีร่วมสมัย'
      };
    case 'ญี่ปุ่นร่วมสมัย':
      return {
        en: 'Contemporary Japan, present day',
        people: 'Japanese',
        outfit: 'modern everyday clothing',
        th: 'ญี่ปุ่นร่วมสมัย'
      };
    case 'ญี่ปุ่นย้อนยุค':
      return {
        en: 'Historical Japan, the exact era follows the script',
        people: 'Japanese',
        outfit: 'traditional Japanese period costume (kimono, hakama)',
        th: 'ญี่ปุ่นย้อนยุค'
      };
    case 'ตะวันตกร่วมสมัย':
      return {
        en: 'Contemporary Western world (Europe or North America), present day',
        people: 'Western',
        outfit: 'modern everyday clothing',
        th: 'ตะวันตกร่วมสมัย'
      };
    case 'ตะวันตกย้อนยุค':
      return {
        en: 'Historical Europe, the exact period follows the script',
        people: 'European',
        outfit: 'European period costume',
        th: 'ตะวันตกย้อนยุค'
      };
    case 'กำหนดเอง':
    default: {
      const isCustom = setting === 'กำหนดเอง';
      const baseEn = isCustom ? (customSetting || '') : setting;
      const baseThai = isCustom ? (customSettingThai || customSetting || 'กำหนดเอง') : setting;
      
      const en = isCustom && customSettingThai ? `${baseEn} (${customSettingThai})` : baseEn;
      const th = baseThai;
      // การค้นหาคำระบุเชื้อชาติ (People) อัตโนมัติจากข้อความกำหนดเอง
      let people = '';
      const searchText = (en + ' ' + th).toLowerCase();
      
      if (searchText.includes('ไทย') || searchText.includes('thai')) people = 'Thai';
      else if (searchText.includes('จีน') || searchText.includes('chinese')) people = 'Chinese';
      else if (searchText.includes('เกาหลี') || searchText.includes('korean')) people = 'Korean';
      else if (searchText.includes('ญี่ปุ่น') || searchText.includes('japanese')) people = 'Japanese';
      else if (searchText.includes('ตะวันตก') || searchText.includes('western') || searchText.includes('europe')) people = 'Western';
      return {
        en,
        people,
        outfit: '', // กำหนดเป็นค่าว่างตามที่ได้รับมอบหมาย
        th
      };
    }
  }
}
