const parseAssetDefinitions = (processed: string): Asset[] | null => {
  if (!processed) return null;
  
  const blockMatch = processed.match(/\[ASSET DEFINITIONS\]([\s\S]*?)\[END ASSET DEFINITIONS\]/i);
  if (!blockMatch) return null;
  const block = blockMatch[1];
  const sectionMap: Array<{ heading: string; type: AssetType; idPrefix: string }> = [
    { heading: 'CHARACTERS', type: 'Character' as AssetType, idPrefix: '@CH' },
    { heading: 'LOCATIONS', type: 'Location' as AssetType, idPrefix: '@LOC' },
    { heading: 'PROPS', type: 'Prop' as AssetType, idPrefix: '@PROP' }
  ];
  const assets: Asset[] = [];
  for (const section of sectionMap) {
    const sectionRegex = new RegExp(`\\[${section.heading}\\]([\\s\\S]*?)(?=\\[(?:CHARACTERS|LOCATIONS|PROPS)\\]|$)`, 'i');
    const sectionMatch = block.match(sectionRegex);
    if (!sectionMatch) continue;
    const sectionText = sectionMatch[1];
    const lines = sectionText.split(/\r?\n/);
    
    let currentAsset: Partial<Asset> & { tempLines: string[] } | null = null;
    const finalizeAsset = (p: Partial<Asset> & { tempLines: string[] }) => {
      const id = p.id || '';
      const fields: Record<string, string> = {};
      let name = '';
      let visualPrompt = '';
      p.tempLines.forEach(line => {
        const colonMatch = line.match(/^([^:]+):\s*(.*)$/);
        if (colonMatch) {
          const key = colonMatch[1].trim().toLowerCase();
          fields[key] = colonMatch[2].trim();
        }
      });
      name = fields['name'] || fields['ชื่อ'] || '';
      visualPrompt = fields['visual prompt'] || '';
      if (p.tempLines.length > 0) {
        const firstLine = p.tempLines[0].trim();
        if (firstLine.includes('|')) {
          const parts = firstLine.split('|').map(s => s.trim()).filter(Boolean);
          if (parts.length >= 1) {
            name = parts[0];
            visualPrompt = parts.slice(1).join(' | ').trim();
          }
        }
      }
      if (!name) return;
      const isNameSuspicious = name.length > 40 || name.includes(',');
      
      console.log(`[${id}] ${name} / ${visualPrompt.substring(0, 60)}${visualPrompt.length > 60 ? '...' : ''}`);
      let metadata: AssetMetadata = {
        appearsInScenes: fields['appears in scenes'] || '',
        evidence: fields['evidence'] || ''
      };
      if (section.type === 'Character') {
        metadata = {
          ...metadata,
          role: fields['role'] || '',
          identityLock: fields['identity lock'] || '',
          age: fields['age'] || '',
          gender: fields['gender'] || '',
          appearance: [fields['face'], fields['appearance'], fields['distinctive features']].filter(Boolean).join(', '),
          hair: fields['hair'] || '',
          body: fields['body'] || '',
          wardrobe: fields['wardrobe'] || '',
          accessories: fields['accessories'] || '',
          personality: fields['personality'] || '',
          mood: fields['default mood'] || ''
        };
      } else if (section.type === 'Location') {
        metadata = {
          ...metadata,
          spatialLock: fields['spatial lock'] || '',
          locationType: fields['type'] || '',
          architecture: fields['architecture'] || '',
          timeOfDay: fields['time of day'] || '',
          lighting: fields['lighting'] || '',
          weather: fields['weather'] || '',
          importantObjects: fields['key objects'] || '',
          countryCulture: fields['country/culture'] || '',
          era: fields['era'] || '',
          layout: fields['layout'] || '',
          materials: fields['materials'] || '',
          environment: fields['environment'] || ''
        };
      } else if (section.type === 'Prop') {
        metadata = {
          ...metadata,
          visualLock: fields['visual lock'] || '',
          category: fields['category'] || '',
          material: fields['material'] || '',
          shape: fields['shape'] || '',
          color: fields['color'] || '',
          size: fields['size'] || '',
          function: fields['function'] || '',
          distinctiveDetails: fields['distinctive details'] || '',
          era: fields['era'] || ''
        };
      }
      const desc = p.tempLines.join('\n');
      assets.push({
        id,
        type: section.type,
        name,
        description: desc,
        metadata,
        visualPrompt,
        status: 'NO IMAGE',
        isLocked: false,
        isNameSuspicious,
        generationOptions: { mode: 'Reference Sheet', aspectRatio: '16:9', quality: 'Standard' }
      });
    };
    lines.forEach(line => {
      const trimmed = line.trim();
      if (!trimmed || trimmed.includes('| ID | Name |') || trimmed.includes('|---|')) return;
      const idMatch = trimmed.match(/^(@(?:CH|LOC|PROP)\d+)/i) || trimmed.match(/^\|\s*(@(?:CH|LOC|PROP)\d+)/i);
      
      if (idMatch) {
        if (currentAsset) finalizeAsset(currentAsset);
        const rawId = idMatch[1].toUpperCase();
        const lineWithoutId = trimmed.replace(idMatch[0], '').trim();
        currentAsset = { id: rawId, type: section.type, tempLines: [lineWithoutId] };
      } else if (currentAsset) {
        currentAsset.tempLines.push(trimmed);
      }
    });
    if (currentAsset) finalizeAsset(currentAsset);
  }
  return assets.length > 0 ? assets : null;
};
const mergeParsedAssets = (existingAssets: Asset[], parsedAssets: Asset[], storyboard: Shot[]): { assets: Asset[]; stats: { new: number; existing: number; missing: number; deleted: number } } => {
  const existing = Array.isArray(existingAssets) ? existingAssets : [];
  const finalList: Asset[] = [];
  const matchedOldIds = new Set<string>();
  const stats = { new: 0, existing: 0, missing: 0, deleted: 0 };
  const normalize = (n: string) => normalizeAssetName(n);
  const findOldMatch = (parsed: Asset) => {
    const pName = normalize(parsed.name);
    return existing.find(old => {
      if (old.type !== parsed.type) return false;
      if (normalize(old.name) === pName) return true;
      const aliases = (old.aliases || '').split(',').map(a => normalize(a)).filter(Boolean);
      return aliases.includes(pName);
    });
  };
  const pendingNew: Asset[] = [];
  // 1. Process parsed assets from script
  parsedAssets.forEach(parsed => {
    const old = findOldMatch(parsed);
    if (old) {
      if (matchedOldIds.has(old.id)) return; // Already matched/merged
      let nextName = parsed.name;
      let nextVisualPrompt = parsed.visualPrompt;
      let nextVisualPromptThai = old.visualPromptThai;
      if (old.nameEditedByUser) nextName = old.name;
      if (old.promptEditedByUser) {
        nextVisualPrompt = old.visualPrompt;
        nextVisualPromptThai = old.visualPromptThai;
      } else if (parsed.visualPrompt !== old.visualPrompt) {
        nextVisualPrompt = parsed.visualPrompt;
        nextVisualPromptThai = undefined;
      }
      finalList.push({
        ...parsed,
        id: old.id,
        name: nextName,
        visualPrompt: nextVisualPrompt,
        visualPromptThai: nextVisualPromptThai,
        metadata: { ...parsed.metadata, identityLock: old.metadata?.identityLock || parsed.metadata?.identityLock, spatialLock: old.metadata?.spatialLock || parsed.metadata?.spatialLock },
        aliases: old.aliases || parsed.aliases,
        voiceProfile: old.voiceProfile || parsed.voiceProfile,
        voiceProfileThai: old.voiceProfileThai || parsed.voiceProfileThai,
        isLocked: old.isLocked,
        nameEditedByUser: old.nameEditedByUser,
        promptEditedByUser: old.promptEditedByUser,
        imageVisualPrompt: old.imageVisualPrompt,
        imageKey: old.imageKey,
        pendingImageKey: old.pendingImageKey,
        status: (old.status === 'READY' || old.status === 'LOCKED' || old.status === 'GENERATING') ? old.status : parsed.status,
        missingFromScript: false,
        addedInStep3: old.addedInStep3,
        referenceMediaId: old.referenceMediaId || parsed.referenceMediaId,
        referenceBase64: old.referenceBase64 || parsed.referenceBase64,
        referenceMimeType: old.referenceMimeType || parsed.referenceMimeType,
        sourceReferences: old.sourceReferences || parsed.sourceReferences,
        generationOptions: old.generationOptions || parsed.generationOptions,
        generationError: old.generationError || parsed.generationError
      });
      matchedOldIds.add(old.id);
      stats.existing++;
    } else {
      pendingNew.push({ ...parsed, missingFromScript: false });
      stats.new++;
    }
  });
  // 2. Identify unmatched existing assets
  existing.forEach(old => {
    if (matchedOldIds.has(old.id)) return;
    // Requirement: Keep assets added in Step 3 always, and don't show missing badge
    if (old.addedInStep3) {
      finalList.push({ ...old, missingFromScript: false });
      stats.existing++;
      return;
    }
    const isUsed = storyboard.some(s => 
      s.locationId === old.id || 
      s.characterIds?.includes(old.id) || 
      s.propIds?.includes(old.id) ||
      s.speakerId === old.id
    );
    const hasImage = !!old.referenceBase64 || !!old.referenceMediaId;
    const isRetained = isUsed || hasImage || old.isLocked || old.status === 'GENERATING';
    if (isRetained) {
      finalList.push({ ...old, missingFromScript: true });
      stats.missing++;
    } else {
      stats.deleted++;
    }
  });
  // 3. Assign IDs to new assets
  const getNextId = (type: AssetType, currentList: Asset[]) => {
    const prefix = type === 'Character' ? '@CH' : type === 'Location' ? '@LOC' : '@PROP';
    const nums = currentList
      .filter(a => a.type === type && a.id.startsWith(prefix))
      .map(a => parseInt(a.id.replace(prefix, '')) || 0);
    const max = nums.length > 0 ? Math.max(...nums) : 0;
    return `${prefix}${(max + 1).toString().padStart(2, '0')}`;
  };
  pendingNew.forEach(pn => {
    const prefix = pn.type === 'Character' ? '@CH' : pn.type === 'Location' ? '@LOC' : '@PROP';
    const taken = finalList.some(a => a.id === pn.id);
    if (!pn.id || !pn.id.startsWith(prefix) || taken) pn.id = getNextId(pn.type, finalList);
    finalList.push(pn);
  });
  return { assets: finalList, stats };
};
const composePromptFromMetadata = (asset: Asset, world: StoryWorld) => {
  const meta = asset.metadata;
  const ethnicityPrefix = world.people || 'Person';
  
  if (asset.type === 'Character') {
    const fields = [meta.age, meta.gender, meta.hair, meta.appearance, meta.body, meta.wardrobe, meta.accessories].filter(Boolean);
    if (fields.length === 0) return asset.name;
    return `${ethnicityPrefix} person, ${fields.join(', ')}`;
  } else if (asset.type === 'Location') {
    const fields = [meta.locationType, meta.architecture, meta.timeOfDay, meta.lighting, meta.materials, meta.importantObjects].filter(Boolean);
    if (fields.length === 0) return asset.name;
    return `${ethnicityPrefix} style setting, ${fields.join(', ')}`;
  } else if (asset.type === 'Prop') {
    const fields = [meta.material, meta.shape, meta.color, meta.size, meta.distinctiveDetails].filter(Boolean);
    if (fields.length === 0) return asset.name;
    return `${ethnicityPrefix} object, ${fields.join(', ')}`;
  }
  return asset.name;
};
const getFinalPrompt = (asset: Asset, world: StoryWorld) => {

  return composePromptFromMetadata(asset, world);

            <p className="font-bold opacity-80 mt-1" style={{ fontSize: 'var(--fs-micro)' }}>รายการที่เตือนอาจได้ภาพที่ไม่ตรงกับเนื้อเรื่องหรือยุคสมัย</p>

        })()}
      </ConfirmDialog>
      {editingAsset && (
        <AssetEditPanel asset={editingAsset} state={state} onClose={() => setEditingAsset(null)} onUpdate={(patch) => { handleUpdateAsset(editingAsset.id, patch); setEditingAsset(prev => prev ? { ...prev, ...patch } : null); }} onGenerate={(a) => { setEditingAsset(null); handleGenerateSingleAsset(a); }} onUploadReference={() => {}} onUploadFinal={(file) => handleUploadFinal(editingAsset, file)} />
      )}
      {previewAsset && previewAsset.referenceBase64 && (
        <div className="fixed inset-0 bg-black/95 z-[200] flex items-center justify-center p-6 backdrop-blur-md" onClick={() => setPreviewAsset(null)}>
          <div className="max-w-5xl w-full flex flex-col gap-4 relative" onClick={e => e.stopPropagation()}>

const AssetCard: React.FC<{ 
  asset: Asset; 
  world: StoryWorld;
  isGenerating: boolean; 
  isMergeMode?: boolean;
  isSelected?: boolean;
  isImageOld: boolean;
  onToggleSelect?: () => void;
  onEdit: () => void; 
  onGenerate: () => void; 
  onUpdate: (patch: Partial<Asset>) => void; 
  onDelete: () => void; 
  onPreview: () => void; 
 }> = ({ asset, world, isGenerating, isMergeMode, isSelected, isImageOld, onToggleSelect, onEdit, onGenerate, onUpdate, onDelete, onPreview }) => {
  const hasBase64 = !!asset.referenceBase64;
  const hasMediaId = !!asset.referenceMediaId;
  const hasAssetRecord = hasBase64 || hasMediaId;
  const [menuOpen, setMenuOpen] = useState(false);
  const menuRef = useRef<HTMLDivElement>(null);
  useEffect(() => {
    if (!menuOpen) return;
    const onDown = (e: MouseEvent) => { if (!menuRef.current?.contains(e.target as Node)) setMenuOpen(false); };
    const onKey = (e: KeyboardEvent) => { if (e.key === 'Escape') setMenuOpen(false); };
    document.addEventListener('mousedown', onDown);
    document.addEventListener('keydown', onKey);
    return () => { document.removeEventListener('mousedown', onDown); document.removeEventListener('keydown', onKey); };
  }, [menuOpen]);
  
  const finalPromptText = getFinalPrompt(asset, world);
  const isPromptEmpty = !finalPromptText || finalPromptText.length < 2;
  
  const refCount = asset.sourceReferences?.length || 0;
  const warningLabel = useMemo(() => {
    if (hasAssetRecord || isGenerating) return null;
    const prompt = finalPromptText;
    if (prompt.length < 30) return "คำบรรยายสั้นเกินไป — ภาพอาจไม่ตรงเรื่อง";
    if (asset.type === 'Character' && !CLOTHING_WORDS.test(prompt)) return "คำบรรยายไม่มีชุด — ภาพอาจไม่ตรงยุค";
    return null;
  }, [hasAssetRecord, isGenerating, finalPromptText, asset.type]);
  return (
    <div 
      onClick={isMergeMode ? onToggleSelect : undefined}
      className={`group bg-[var(--ui-surface)] border rounded-2xl overflow-hidden p-5 space-y-4 transition-all duration-300 relative shadow-lg ${isMergeMode ? 'cursor-pointer' : ''} ${asset.isLocked && !isMergeMode ? 'border-[var(--ui-primary)]' : isSelected ? 'border-[var(--ui-primary)] ring-4 ring-[var(--ui-primary)]/20 shadow-lg' : 'border-[var(--ui-border)]'}`}
    >
      <div className="flex justify-between items-start">
        <div className="flex flex-col min-w-0 flex-1">
          <div className="flex items-center gap-2">
            {isMergeMode && (
              <div className={`w-5 h-5 rounded-full border-2 flex items-center justify-center shrink-0 ${isSelected ? 'bg-[var(--ui-primary)] border-[var(--ui-primary)]' : 'border-[var(--ui-text-muted)]'}`}>
                {isSelected && <span className="material-symbols-outlined text-white text-[14px] font-bold">check</span>}
              </div>
            )}
            <h3 style={{ fontSize: 'var(--fs-body)' }} className={`font-extrabold text-[var(--ui-text-primary)] truncate uppercase ${asset.isNameSuspicious ? 'text-amber-500' : ''}`}>{asset.name}</h3>
          </div>
          <div className="flex items-center gap-1.5 flex-wrap relative">
            <span style={{ fontSize: 'var(--fs-micro)' }} className="text-[var(--ui-text-muted)] font-mono font-bold uppercase">{asset.id}</span>
            {warningLabel && (
              <div className="group/warn relative">
                <span className="bg-yellow-500 text-black text-[10px] font-black px-1.5 py-0.5 rounded cursor-help">
                  {warningLabel.split(' — ')[0]}
                </span>
                <div className="absolute left-0 bottom-full mb-2 hidden group-hover/warn:block w-48 bg-black text-white p-2 rounded-lg text-[10px] font-bold shadow-xl z-50 animate-in fade-in zoom-in-95">
                  {warningLabel}
                </div>
              </div>
            )}
            {asset.missingFromScript && (
              <span className="bg-slate-500/20 text-slate-400 text-[9px] font-black px-1.5 py-0.5 rounded border border-slate-500/30">ไม่พบในบทล่าสุด</span>
            )}
            {refCount > 0 && (
              <span className="bg-[var(--ui-accent)]/10 text-[var(--ui-accent)] text-[9px] font-black px-1.5 py-0.5 rounded border border-[var(--ui-accent)]/20">อ้างอิง {refCount} รูป</span>
            )}
            {asset.isNameSuspicious && (
              <span className="material-symbols-outlined text-[14px] text-amber-500" title="ชื่ออาจผิดปกติ (ยาวเกินไปหรือมีคอมมา)">warning</span>
            )}
          </div>
        </div>
        {!isMergeMode && (
          <div className="flex items-center gap-1">
            <button onClick={(e) => { e.stopPropagation(); if(hasAssetRecord) onUpdate({ isLocked: !asset.isLocked }); }} disabled={!hasAssetRecord} className={`p-1.5 rounded-lg ${asset.isLocked ? 'text-[var(--ui-primary)]' : 'text-[var(--ui-text-muted)] opacity-40 hover:opacity-100'}`}>
              <span className="material-symbols-outlined text-[20px] font-bold">{asset.isLocked ? 'lock' : 'lock_open'}</span>
            </button>
            <div className="relative" ref={menuRef}>
              <button onClick={(e) => { e.stopPropagation(); setMenuOpen(!menuOpen); }} title="คำสั่ง" className="text-[var(--ui-text-muted)] hover:text-[var(--ui-text-primary)]"><span className="material-symbols-outlined text-2xl font-bold">more_vert</span></button>
              {menuOpen && (
                <div className="absolute right-0 top-full mt-1 z-50 w-44 bg-[var(--ui-surface-raised)] border border-[var(--ui-border)] rounded-xl shadow-2xl overflow-hidden py-1">
                  <button onClick={(e) => { e.stopPropagation(); setMenuOpen(false); onEdit(); }} className="w-full text-left px-4 py-2.5 text-xs font-bold text-[var(--ui-text-primary)] hover:bg-[var(--ui-surface-soft)] flex items-center gap-2"><span className="material-symbols-outlined text-sm">edit</span>แก้ไข</button>
                  {!asset.isLocked && !isGenerating && !isPromptEmpty && (
                    <button onClick={(e) => { e.stopPropagation(); setMenuOpen(false); onGenerate(); }} className="w-full text-left px-4 py-2.5 text-xs font-bold text-[var(--ui-text-primary)] hover:bg-[var(--ui-surface-soft)] flex items-center gap-2"><span className="material-symbols-outlined text-sm">refresh</span>{hasAssetRecord ? 'สร้างภาพใหม่' : 'สร้างภาพ'}</button>
                  )}
                  {!asset.isLocked && !isGenerating && (
                    <button onClick={(e) => { e.stopPropagation(); setMenuOpen(false); onDelete(); }} className="w-full text-left px-4 py-2.5 text-xs font-bold text-[var(--ui-danger)] hover:bg-[var(--ui-surface-soft)] flex items-center gap-2"><span className="material-symbols-outlined text-sm">delete</span>ลบ</button>
                  )}
                </div>
              )}
            </div>
          </div>
        )}
      </div>
      <div className="aspect-video bg-[#C9C9C9] rounded-xl relative overflow-hidden" onClick={!isMergeMode && hasBase64 && !isGenerating ? onPreview : undefined}>
        {hasBase64 ? (
          <img src={`data:${asset.referenceMimeType || 'image/png'};base64,${asset.referenceBase64}`} className={`w-full h-full object-contain ${isGenerating ? 'opacity-40' : ''}`} />
        ) : hasMediaId ? (
          <div className="w-full h-full flex flex-col items-center justify-center bg-slate-800 text-center p-4">
             <span className="material-symbols-outlined text-amber-500 text-3xl mb-2">cloud_done</span>
             <p className="text-[10px] text-white font-bold leading-tight">ภาพอยู่ในคลัง Flow</p>
             <p className="text-[9px] text-slate-400 font-medium mt-1">ระบบกำลังโหลดภาพ — ถ้าไม่ขึ้นให้กด เปิดไฟล์ ที่แถบซ้าย</p>
          </div>
        ) : (
          <div className="w-full h-full flex flex-col items-center justify-center opacity-40">
             <span className="material-symbols-outlined text-3xl mb-2">add_photo_alternate</span>
             {!isMergeMode && <button onClick={(e) => { e.stopPropagation(); onGenerate(); }} disabled={isPromptEmpty} className="px-6 py-2.5 bg-[var(--ui-accent)] text-black font-extrabold rounded-lg text-[11px] disabled:opacity-30">สร้างภาพ</button>}
          </div>
        )}
        
        {isImageOld && !isGenerating && !isMergeMode && (
          <div className="absolute top-2 right-2 bg-amber-500 text-black px-2 py-1 rounded text-[9px] font-black uppercase shadow-lg">
            ภาพเก่า — กดสร้างใหม่
          </div>
        )}
        {isGenerating && (
          <div className="absolute inset-0 bg-black/40 flex flex-col items-center justify-center z-30 backdrop-blur-[2px]">
             <div className="w-8 h-8 border-[3px] border-[var(--ui-accent)] border-t-transparent rounded-full animate-spin mb-3" />
             <span style={{ fontSize: '10px' }} className="text-[var(--ui-accent)] font-extrabold animate-pulse">กำลังสร้าง...</span>
          </div>
        )}
      </div>
      <div className="flex flex-col gap-2">
        <p style={{ fontSize: 'var(--fs-small)' }} className="text-[var(--ui-text-muted)] font-medium line-clamp-2">{finalPromptText || "ยังไม่มีรายละเอียด..."}</p>
        {asset.status === 'FAILED' && <p className="text-[11px] font-bold text-[var(--ui-danger)]" title={asset.generationError}>สร้างภาพไม่สำเร็จ — {friendlyError(asset.generationError)}</p>}
      </div>
    </div>
  );
};
