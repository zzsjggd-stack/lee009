}
export const NextStepButton: React.FC<Props> = ({ 
  onNext, 
  missingRequirements, 
  label = "ขั้นตอนถัดไป",
  className = "" 
}) => {
  const [showPopup, setShowPopup] = useState(false);
  const timerRef = useRef<number | null>(null);
  const containerRef = useRef<HTMLDivElement>(null);
  const hasMissing = missingRequirements.length > 0;
  // Auto-hide popup if requirements are met while it's open
  useEffect(() => {
    if (!hasMissing && showPopup) {
      setShowPopup(false);
    }
  }, [hasMissing, showPopup]);
  const handleShowPopup = () => {
    if (!hasMissing) return;
    // Clear existing timer to restart the 3s window
    if (timerRef.current) {
      window.clearTimeout(timerRef.current);
    }
    setShowPopup(true);
    timerRef.current = window.setTimeout(() => {
      setShowPopup(false);
    }, 3000);
  };
  const handleClick = (e: React.MouseEvent) => {
    if (hasMissing) {
      e.preventDefault();
      handleShowPopup();
    } else {
      onNext();
    }
  };
  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === 'Enter' || e.key === ' ') {
      if (hasMissing) {
        e.preventDefault();
        handleShowPopup();
      }
    }
  };
  // Cleanup timer on unmount
  useEffect(() => {
    return () => {
      if (timerRef.current) window.clearTimeout(timerRef.current);
    };
  }, []);
  return (
    <div ref={containerRef} className="relative inline-block">
      {/* Validation Popup */}
      {showPopup && hasMissing && (
        <div 
          role="status"
          aria-live="polite"
          className="absolute bottom-full left-1/2 -translate-x-1/2 mb-3.5 w-[280px] md:w-[320px] bg-[var(--ui-surface-raised)] border border-[var(--ui-accent)]/40 p-4 rounded-xl shadow-[0_12px_35px_rgba(0,0,0,0.35)] z-[100] animate-in fade-in slide-in-from-bottom-2 duration-200 pointer-events-none"
        >
          <div className="flex items-start gap-3">
            <span className="material-symbols-outlined text-[var(--ui-accent)] text-xl shrink-0 mt-0.5">info</span>
            <div className="flex flex-col gap-1.5">
              <h4 className="text-[13px] font-bold text-[var(--ui-text-primary)] leading-tight">ยังดำเนินการต่อไม่ได้</h4>
              <p className="text-[11px] font-medium text-[var(--ui-text-secondary)]">กรุณาทำรายการต่อไปนี้ให้ครบ</p>
              
              <ul className="mt-1 space-y-1">
                {missingRequirements.slice(0, 3).map((req, idx) => (
                  <li key={idx} className="flex items-center gap-2 text-[11px] font-bold text-[var(--ui-text-secondary)]">
                    <span className="w-1 h-1 rounded-full bg-[var(--ui-accent)] opacity-60" />
                    {req}
                  </li>
                ))}
                {missingRequirements.length > 3 && (
                  <li className="text-[10px] font-bold text-[var(--ui-text-muted)] pl-3">
                    + อีก {missingRequirements.length - 3} รายการ
                  </li>
                )}
              </ul>
            </div>
          </div>
          
          {/* Subtle Caret pointing to button */}
          <div className="absolute bottom-0 left-1/2 -translate-x-1/2 translate-y-1/2 w-3 h-3 bg-[var(--ui-surface-raised)] border-b border-r border-[var(--ui-accent)]/40 rotate-45" />
        </div>
      )}
      {/* The Guarded Button */}
      <button
        onClick={handleClick}
        onKeyDown={handleKeyDown}
        aria-disabled={hasMissing}
        className={`g2-cta relative transition-all flex items-center justify-center gap-2 outline-none group ${className} ${
          hasMissing 
            ? 'opacity-40 cursor-default grayscale-[0.2]' 
            : 'hover:bg-[var(--ui-accent-hover)] active:scale-95 shadow-xl'
        }`}
      >
        {label}
        <span className="material-symbols-outlined font-bold transition-transform group-hover:translate-x-0.5">arrow_forward</span>
        
        {/* Subtle highlight when requirement is missing and clicked */}
        {showPopup && (
          <div className="absolute inset-0 rounded-inherit border border-amber-500/25 pointer-events-none" />
        )}
      </button>
    </div>
  );
};
