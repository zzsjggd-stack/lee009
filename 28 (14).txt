import { AppState } from '../types';
/**
 * ตัดข้อมูล Base64 ขนาดใหญ่ออกจาก Object ก่อนบันทึก เพื่อป้องกัน QuotaExceededError
 */
export const sanitizeStateForStorage = (state: AppState): AppState => {
  return {
    ...state,
    // ลบภาพพรีวิวตัวละคร
    assets: state.assets.map(asset => ({
      ...asset,
      referenceBase64: undefined,
      sourceReferences: asset.sourceReferences?.map(ref => ({
        ...ref,
        base64: undefined
      }))
    })),
    // ลบภาพพรีวิวช็อต
    storyboard: state.storyboard.map(shot => ({
      ...shot,
      firstFrameBase64: undefined
    })),
    // ลบวิดีโอพรีวิว
    videos: state.videos.map(video => ({
      ...video,
      videoBase64: undefined
    }))
  };
};
/**
 * บันทึกลง SessionStorage พร้อมระบบกู้คืนเมื่อพื้นที่เต็ม
 */
export const safeSessionSave = (key: string, data: any) => {
  try {
    sessionStorage.setItem(key, JSON.stringify(data));
    return true;
  } catch (error) {
    try {
      // ลองตัดประวัติออกก่อนถ้าพื้นที่เต็ม
      const slim = { ...data, script: { ...(data?.script || {}), history: [] } };
      sessionStorage.setItem(key, JSON.stringify(slim));
      console.warn('Storage almost full — script history removed');
      return true;
    } catch (error2) {
      console.error(`Autosave failed [${key}] — Please save ZIP`, error2);
      window.dispatchEvent(new CustomEvent('G2_AUTOSAVE_FAILED'));
      return false;
    }
  }
};
