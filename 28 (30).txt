import React, { useState } from 'react';
import { friendlyError } from '../lib/friendlyError';
export const ErrorDetails: React.FC<{ raw?: string; className?: string }> = ({ raw, className = '' }) => {
  const [open, setOpen] = useState(false);
  if (!raw) return null;
  const friendly = friendlyError(raw);
  return (
    <div className={className}>
      <p className="text-xs font-bold">{friendly}</p>
      {friendly !== raw && (
        <>
          <button type="button" onClick={() => setOpen(!open)} className="mt-1 text-[11px] font-bold underline opacity-70 hover:opacity-100">
            {open ? 'ซ่อนรายละเอียด' : 'ดูรายละเอียด'}
          </button>
          {open && <p className="mt-1 text-[11px] font-mono break-all whitespace-pre-wrap opacity-70 select-text">{raw}</p>}
        </>
      )}
    </div>
  );
};
