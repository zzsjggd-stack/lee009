import React, { useEffect } from 'react';
type Tone = 'default' | 'danger' | 'warning';
interface ConfirmDialogProps {
  open: boolean;
  title: string;
  message?: React.ReactNode;
  icon?: string;
  tone?: Tone;
  confirmLabel?: string;
  cancelLabel?: string;
  onConfirm: () => void;
  onCancel: () => void;
  confirmDisabled?: boolean;
  busy?: boolean;
  extraActions?: { label: string; onClick: () => void; tone?: Tone }[];
  children?: React.ReactNode;
}
export const ConfirmDialog: React.FC<ConfirmDialogProps> = ({
  open, title, message, icon, tone = 'default',
  confirmLabel = 'ยืนยัน', cancelLabel = 'ยกเลิก',
  onConfirm, onCancel, confirmDisabled, busy, extraActions, children
}) => {
  useEffect(() => {
    if (!open) return;
    const onKey = (e: KeyboardEvent) => { 
      if (e.key === 'Escape' && !busy) onCancel(); 
    };
    document.addEventListener('keydown', onKey);
    return () => document.removeEventListener('keydown', onKey);
  }, [open, busy, onCancel]);
  if (!open) return null;
  const iconClass = tone === 'danger'
    ? 'bg-[var(--ui-danger)]/10 text-[var(--ui-danger)]'
    : tone === 'warning'
      ? 'bg-[var(--ui-accent)]/10 text-[var(--ui-accent)]'
      : 'bg-[var(--ui-primary)]/10 text-[var(--ui-primary)]';
      
  const confirmClass = tone === 'danger'
    ? 'bg-[var(--ui-danger)] text-white hover:opacity-90'
    : 'g2-cta';
    
  const btnStyle = { minHeight: 'var(--ui-control-height)', fontSize: 'var(--fs-body)' };
  return (
    <div
      className="fixed inset-0 z-[300] bg-black/70 backdrop-blur-sm flex items-center justify-center p-6"
      onMouseDown={(e) => { if (e.target === e.currentTarget && !busy) onCancel(); }}
    >
      <div
        role="dialog"
        aria-modal="true"
        className="w-full max-w-md bg-[var(--ui-surface-raised)] border border-[var(--ui-border)] rounded-3xl shadow-2xl text-center flex flex-col"
        style={{ padding: 'var(--ui-panel-pad)', gap: 'calc(var(--ui-gap) * 0.6)' }}
      >
        {icon && (
          <div className={`w-14 h-14 rounded-full flex items-center justify-center mx-auto ${iconClass}`}>
            <span className="material-symbols-outlined text-3xl font-bold">{icon}</span>
          </div>
        )}
        
        <h3 className="font-extrabold text-[var(--ui-text-primary)] tracking-tight" style={{ fontSize: 'var(--fs-h2)' }}>
          {title}
        </h3>
        
        {message && (
          <div className="text-[var(--ui-text-secondary)] leading-relaxed" style={{ fontSize: 'var(--fs-body)' }}>
            {message}
          </div>
        )}
        
        {children}
        
        <div className="flex flex-col gap-3 pt-1">
          <button 
            type="button" 
            onClick={onConfirm} 
            disabled={confirmDisabled || busy}
            className={`w-full rounded-2xl font-extrabold shadow-lg transition-all disabled:opacity-40 flex items-center justify-center gap-2 ${confirmClass}`} 
            style={btnStyle}
          >
            {busy && <span className="w-4 h-4 border-2 border-current border-t-transparent rounded-full animate-spin" />}
            {confirmLabel}
          </button>
          
          {extraActions?.map((a, i) => (
            <button 
              key={i} 
              type="button" 
              onClick={a.onClick} 
              disabled={busy}
              className={`w-full rounded-2xl font-bold border transition-all disabled:opacity-40 ${
                a.tone === 'danger'
                  ? 'border-[var(--ui-danger)]/30 text-[var(--ui-danger)] bg-[var(--ui-danger)]/10'
                  : 'border-[var(--ui-border)] text-[var(--ui-text-primary)] bg-[var(--ui-surface-soft)]'
              }`} 
              style={btnStyle}
            >
              {a.label}
            </button>
          ))}
          
          <button 
            type="button" 
            onClick={onCancel} 
            disabled={busy}
            className="w-full rounded-2xl font-bold text-[var(--ui-text-muted)] hover:text-[var(--ui-text-primary)] transition-colors disabled:opacity-40" 
            style={btnStyle}
          >
            {cancelLabel}
          </button>
        </div>
      </div>
    </div>
  );
};
