import { ScriptData } from '../types';
/**
 * Interface for resolved language data
 */
export interface ResolvedLanguage {
  en: string;      // English description for generation/AI prompts
  th: string;      // Short Thai name for UI display
  isThai: boolean; // Whether the language is considered Thai or a Thai dialect
  isMulti: boolean; // Whether it's a multilingual setting
}
/**
 * Resolves the spoken language based on script settings.
 */
export function resolveSpokenLanguage(script: ScriptData): ResolvedLanguage {
  const { spokenLanguage, setting, customSetting, customSpokenLanguage, customSpokenLanguageEn } = script;
  const result: ResolvedLanguage = {
    en: "Central Thai",
    th: "ไทยกลาง",
    isThai: true,
    isMulti: false
  };
  switch (spokenLanguage) {
    case 'ไทยกลาง (มาตรฐาน)':
      return { 
        en: "Thai (Central Thai, standard Bangkok accent)", 
        th: "ไทยกลาง", 
        isThai: true, 
        isMulti: false 
      };
    case 'ไทยใต้ (สำเนียงใต้)':
      return { 
        en: "Southern Thai dialect (Pak Tai), southern accent", 
        th: "ใต้", 
        isThai: true, 
        isMulti: false 
      };
    case 'ไทยคำเมือง (เหนือ)':
      return { 
        en: "Northern Thai dialect (Kham Mueang / Lanna), northern accent", 
        th: "เหนือ", 
        isThai: true, 
        isMulti: false 
      };
    case 'ไทยอีสาน':
      return { 
        en: "Isan dialect (Northeastern Thai), Isan accent", 
        th: "อีสาน", 
        isThai: true, 
        isMulti: false 
      };
    case 'หลายภาษา (Multilingual)':
      return { 
        en: "each character speaks their own native language", 
        th: "หลายภาษา", 
        isThai: false, 
        isMulti: true 
      };
    case 'กำหนดเอง (Custom)':
      const customEn = customSpokenLanguageEn || customSpokenLanguage || "Custom language";
      const customTh = customSpokenLanguage || "กำหนดเอง";
      return { 
        en: customEn, 
        th: customTh, 
        isThai: customTh.includes('ไทย'), 
        isMulti: false 
      };
    case 'อัตโนมัติตามโลกของเรื่อง':
    default:
      // Automatic detection based on story setting
      const activeSetting = setting === 'กำหนดเอง' ? (customSetting || '') : setting;
      
      if (activeSetting.includes('ไทย')) {
        return { en: "Central Thai", th: "ไทยกลาง", isThai: true, isMulti: false };
      } else if (activeSetting.includes('จีน')) {
        return { en: "Mandarin Chinese", th: "จีนกลาง", isThai: false, isMulti: false };
      } else if (activeSetting.includes('เกาหลี')) {
        return { en: "Korean", th: "เกาหลี", isThai: false, isMulti: false };
      } else if (activeSetting.includes('ญี่ปุ่น')) {
        return { en: "Japanese", th: "ญี่ปุ่น", isThai: false, isMulti: false };
      } else if (activeSetting.includes('ตะวันตก')) {
        return { en: "English", th: "อังกฤษ", isThai: false, isMulti: false };
      } else {
        // Fallback search in customSetting keywords if standard options didn't match
        const lowerCustom = activeSetting.toLowerCase();
        if (lowerCustom.includes('ไทย')) return { en: "Central Thai", th: "ไทยกลาง", isThai: true, isMulti: false };
        if (lowerCustom.includes('จีน')) return { en: "Mandarin Chinese", th: "จีนกลาง", isThai: false, isMulti: false };
        if (lowerCustom.includes('เกาหลี')) return { en: "Korean", th: "เกาหลี", isThai: false, isMulti: false };
        if (lowerCustom.includes('ญี่ปุ่น')) return { en: "Japanese", th: "ญี่ปุ่น", isThai: false, isMulti: false };
        if (lowerCustom.includes('อังกฤษ')) return { en: "English", th: "อังกฤษ", isThai: false, isMulti: false };
        
        // Default fallback
        return { en: "Central Thai", th: "ไทยกลาง", isThai: true, isMulti: false };
      }
  }
}
/**
 * Resolves the script/prompt language.
 */
export function resolveScriptLanguage(script: ScriptData): 'Thai' | 'English' {
  return script.language === 'English' ? 'English' : 'Thai';
}
