export interface AssetLine { name: string; desc: string; }
const HEADER_NAMES = new Set(['name', 'ชื่อ', 'item', 'items', 'character', 'characters', 'location', 'locations', 'prop', 'props']);
export function parseAssetLines(text: string): AssetLine[] {
  const out: AssetLine[] = [];
  if (typeof text !== 'string' || !text.trim()) return out;
  for (const rawLine of text.split(/\r?\n/)) {
    let line = rawLine.trim();
    if (!line) continue;
    if (/^[|\-:\s=]+$/.test(line)) continue;
    line = line.replace(/\*\*/g, '');
    line = line.replace(/^\|+|\|+$/g, '').trim();
    line = line.replace(/^(?:[#\-\*\u2022]+\s*|\d+[\.\)]\s*)+/, '').trim();
    line = line.replace(/^@?(?:CH|LOC|PROP)\d+\s*[|:\-—–]?\s*/i, '').trim();
    if (!line) continue;
    let parts: string[] = [];
    if (line.includes('|')) {
      parts = line.split('|');
    } else {
      const sep = [' — ', ' – ', ' - '].find(s => line.includes(s));
      if (sep) {
        const i = line.indexOf(sep);
        parts = [line.slice(0, i), line.slice(i + sep.length)];
      } else {
        const i = line.indexOf(':');
        if (i > 0 && i <= 40) parts = [line.slice(0, i), line.slice(i + 1)];
      }
    }
    parts = parts.map(p => p.trim()).filter(Boolean);
    if (parts.length < 2) continue;
    const name = parts[0].replace(/^["'“”‘’«»]+|["'“”‘’«»]+$/g, '').trim();
    const desc = parts.slice(1).join(', ').trim();
    if (!name || !desc || name.length > 60) continue;
    if (HEADER_NAMES.has(name.toLowerCase())) continue;
    out.push({ name, desc });
  }
  return out;
}
