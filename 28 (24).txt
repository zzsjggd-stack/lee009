/** ข้อความมุมกล้องสำหรับภาพนิ่ง STEP 4 */
export function getCameraFrameText(shot: Shot): string | undefined {
  const s: any = shot;
  if (s.angleId === 'custom') return s.angleCustom || undefined;
  if (s.angleId) return CAMERA_ANGLES.find(a => a.id === s.angleId)?.frame;
  const legacy = s.cameraPreset ? LEGACY_PRESET_MAP[s.cameraPreset] : undefined;
  if (legacy?.angleId) return CAMERA_ANGLES.find(a => a.id === legacy.angleId)?.frame;
  if (s.cameraPreset === 'custom') return s.cameraCustom || undefined;
  return undefined;
}
/** ข้อความการเคลื่อนไหวสำหรับวิดีโอ STEP 5 */
export function getCameraMoveText(shot: Shot): string | undefined {
  const s: any = shot;
  if (s.moveId === 'custom') return s.moveCustom || undefined;
  if (s.moveId) return CAMERA_MOVES.find(m => m.id === s.moveId)?.move;
  const legacy = s.cameraPreset ? LEGACY_PRESET_MAP[s.cameraPreset] : undefined;
  if (legacy) return CAMERA_MOVES.find(m => m.id === legacy.moveId)?.move;
  if (s.cameraPreset === 'custom') return s.cameraCustom || undefined;
  return undefined;
}
