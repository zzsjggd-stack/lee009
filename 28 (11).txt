/**
 * ระบบคัดลอกข้อความ (Clipboard Utility) สำหรับ G2
 * ปรับปรุงให้รองรับสภาพแวดล้อม Sandboxed Iframe ของ Flow โดยเฉพาะ
 */
export async function copyToClipboard(text: string): Promise<boolean> {
  if (!text) return false;
  // 1. ลองใช้ Clipboard API (Modern)
  try {
    if (typeof navigator !== 'undefined' && navigator.clipboard && typeof navigator.clipboard.writeText === 'function') {
      await navigator.clipboard.writeText(text);
      return true;
    }
  } catch (err) {
    // ข้ามไปใช้ Fallback หากติด Permissions Policy (Disallowed)
    console.warn('Modern Clipboard API failed or disallowed:', err);
  }
  // 2. ระบบสำรอง (Fallback): ใช้ Textarea ชั่วคราว
  try {
    const textArea = document.createElement("textarea");
    textArea.value = text;
    
    // ตั้งค่าให้อยู่นอกหน้าจอและมองไม่เห็น
    textArea.style.position = "fixed";
    textArea.style.left = "-9999px";
    textArea.style.top = "-9999px";
    textArea.style.opacity = "0";
    textArea.style.pointerEvents = "none";
    
    document.body.appendChild(textArea);
    textArea.focus();
    textArea.select();
    
    // execCommand เป็น synchronous และมักทำงานได้ใน sandbox หากเกิดจาก user gesture
    const successful = document.execCommand('copy');
    document.body.removeChild(textArea);
    
    return successful;
  } catch (err) {
    console.error('Fallback copy failed:', err);
    return false;
  }
}
