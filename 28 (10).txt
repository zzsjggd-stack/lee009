import React, { useEffect, useRef } from 'react';
import { UITheme } from '../lib/uiPreferences';
interface VisualEffectsProps {
  theme?: UITheme;
}
export const VisualEffects: React.FC<VisualEffectsProps> = ({ theme = 'dark' }) => {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const requestRef = useRef<number>(0);
  const particles = useRef<any[]>([]);
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;
    // Handle Resize
    const handleResize = () => {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
      initParticles();
    };
    // Initialize Particles
    const initParticles = () => {
      const count = 20;
      particles.current = Array.from({ length: count }, () => ({
        x: Math.random() * canvas.width,
        y: Math.random() * canvas.height,
        size: 1 + Math.random() * 2,
        speed: 0.2 + Math.random() * 0.5,
        opacity: 0.03 + Math.random() * 0.07,
        drift: (Math.random() - 0.5) * 0.2,
      }));
    };
    // Animation Logic
    const animate = () => {
      if (document.hidden) {
        requestRef.current = requestAnimationFrame(animate);
        return;
      }
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      // Draw Particles (Dust)
      particles.current.forEach((p) => {
        ctx.beginPath();
        ctx.fillStyle = `rgba(255, 255, 255, ${p.opacity})`;
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();
        // Update Position
        p.y -= p.speed;
        p.x += p.drift;
        // Wrap around
        if (p.y < -10) {
          p.y = canvas.height + 10;
          p.x = Math.random() * canvas.width;
        }
        if (p.x < -10) p.x = canvas.width + 10;
        if (p.x > canvas.width + 10) p.x = -10;
      });
      requestRef.current = requestAnimationFrame(animate);
    };
    window.addEventListener('resize', handleResize);
    handleResize();
    requestRef.current = requestAnimationFrame(animate);
    return () => {
      window.removeEventListener('resize', handleResize);
      cancelAnimationFrame(requestRef.current);
    };
  }, []);
  return (
    <>
      <div className="fixed inset-0 w-full h-full pointer-events-none overflow-hidden select-none z-0">
        <div className={`absolute inset-0 reduced-motion:animate-none bg-[#080B12]`}>
          <div className="blob blob-1 absolute w-[100vw] h-[100vh] rounded-full blur-[120px] opacity-25 bg-[#2F6BF0] -top-1/4 -left-1/4" />
          <div className="blob blob-2 absolute w-[100vw] h-[100vh] rounded-full blur-[120px] opacity-20 bg-[#0EA5E9] -bottom-1/4 -right-1/4" />
          <div className="blob blob-3 absolute w-[80vw] h-[80vh] rounded-full blur-[100px] opacity-20 bg-[#1E3A8A] top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2" />
        </div>
        <canvas 
          ref={canvasRef} 
          className="absolute inset-0 w-full h-full opacity-100 pointer-events-none"
        />
      </div>
      <style dangerouslySetInnerHTML={{ __html: `
        @keyframes blobFloat1 {
          0%, 100% { transform: translate(0, 0) scale(1); }
          33% { transform: translate(10%, 5%) scale(1.1); }
          66% { transform: translate(-5%, 12%) scale(0.9); }
        }
        @keyframes blobFloat2 {
          0%, 100% { transform: translate(0, 0) scale(1.1); }
          33% { transform: translate(-12%, -8%) scale(0.9); }
          66% { transform: translate(8%, -15%) scale(1.05); }
        }
        @keyframes blobFloat3 {
          0%, 100% { transform: translate(-50%, -50%) scale(1); }
          33% { transform: translate(-45%, -55%) scale(1.1); }
          66% { transform: translate(-55%, -48%) scale(0.95); }
        }
        @media (prefers-reduced-motion: reduce) {
          .blob-1, .blob-2, .blob-3 {
            animation: none !important;
          }
        }
      `}} />
    </>
  );
};
