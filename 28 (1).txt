const normalizeStyle = (config: ConfigData): ConfigData => {
  const s = config?.style;
  if (!s || s === 'Custom' || s in STYLES_PROMPT) return config;
  const inner = s.match(/\(([^)]+)\)\s*$/)?.[1]?.trim();
  return inner && inner in STYLES_PROMPT ? { ...config, style: inner } : config;
};

      const actualPatch = typeof patch === 'function' ? patch(prev) : patch;

    });

  // รับผลงานจากคิวทุกหน้า ที่นี่ที่เดียว
  useEffect(() => {

      const id = String(jobId).split('_')[1];
      if (!id) return;
      if (jobId.startsWith('asset_')) {
        patchState(prev => ({
          assets: (prev.assets || []).map(a => a.id === id ? {
            ...a, status: 'READY', referenceBase64: result.base64, referenceMediaId: result.mediaId,
            referenceMimeType: result.mimeType, imageVisualPrompt: a.visualPrompt, generationError: undefined,
            imageKey: a.pendingImageKey ?? a.imageKey, pendingImageKey: undefined
          } : a)
        }));
      }
      if (jobId.startsWith('shot_')) {
        patchState(prev => ({
          storyboard: (prev.storyboard || []).map(s => s.id === id ? {
            ...s, firstFrameMediaId: result.mediaId, firstFrameBase64: result.base64,

            frameSourceKey: s.pendingFrameKey ?? s.frameSourceKey, pendingFrameKey: undefined,
            frameRefIds: s.pendingRefIds ?? s.frameRefIds, pendingRefIds: undefined,

          } : s)
        }));

          };
        });

      if (!id) return;

    const toState = (p: any): AppState => ({

'bg-[var(--ui-primary)] border-[var(--ui-primary)] text-white' : 'bg-transparent border-[var(--ui-border)] text-[var(--ui-text-muted)] hover:border-[var(--ui-border-hover)] hover:text-[var(--ui-text-primary)]'}`}>

                {showSettings && (
                  <React.Suspense fallback={null}>
                    <UISettingsPanel preferences={uiPrefs} onUpdate={setUiPrefs} onClose={() => setShowSettings(false)} />
                  </React.Suspense>
                )}
              </div>
            </header>
            <div className="flex-1 p-[var(--ui-page-pad)]">
              <div className="mx-auto h-full flex flex-col max-w-[1500px]">
                <React.Suspense fallback={<div className="flex items-center justify-center py-32 text-[var(--ui-text-muted)] font-bold">กำลังเปิดขั้นตอน...</div>}>
                  {renderStep()}
                </React.Suspense>
              </div>
            </div>
          </main>
        </div>
        <QueuePanel />
        {autosaveFailed && (
          <div className="fixed top-3 left-1/2 -translate-x-1/2 -translate-y-0 z-[500] flex items-center gap-3 px-5 py-3 bg-[var(--ui-danger)] text-white rounded-xl shadow-2xl">
            <span className="material-symbols-outlined font-bold">warning</span>
            <span className="font-bold" style={{ fontSize: 'var(--fs-small)' }}>บันทึกอัตโนมัติไม่สำเร็จ — กด "บันทึก" ที่แถบซ้ายเก็บไฟล์ ZIP ไว้ก่อน</span>
            <button onClick={() => setAutosaveFailed(false)} className="material-symbols-outlined">close</button>
          </div>
        )}
        {hydrate && (
          <div className="fixed bottom-24 left-1/2 -translate-x-1/2 z-[450] flex items-center gap-3 px-4 py-2 rounded-full bg-[var(--ui-surface-raised)] border border-[var(--ui-border)] shadow-2xl">
            {hydrate.busy
              ? <div className="w-4 h-4 border-2 border-[var(--ui-accent)] border-t-transparent rounded-full animate-spin" />
              : <span className="material-symbols-outlined text-[var(--ui-accent)]">info</span>}
            <span className="font-bold text-[var(--ui-text-primary)]" style={{ fontSize: 'var(--fs-small)' }}>{hydrate.text}</span>
            {!hydrate.busy && <button onClick={() => setHydrate(null)} className="material-symbols-outlined text-[var(--ui-text-muted)] text-xs">close</button>}
          </div>
        )}
        {importStatus && (
          <div className="fixed inset-0 z-[400] bg-black/70 backdrop-blur-sm flex flex-col items-center justify-center gap-4">
            <div className="w-12 h-12 border-4 border-[var(--ui-accent)] border-t-transparent rounded-full animate-spin" />
            <p className="font-bold text-white" style={{ fontSize: 'var(--fs-body)' }}>{importStatus}</p>
          </div>
        )}
      </div>
    </QueueProvider>
  );
}
