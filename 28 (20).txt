  onClose: () => void;
}
export const UISettingsPanel: React.FC<UISettingsPanelProps> = ({ preferences, onUpdate, onClose }) => {
  const panelRef = useRef<HTMLDivElement>(null);
  useEffect(() => {
    const handleClickOutside = (event: MouseEvent) => {
      if (panelRef.current && !panelRef.current.contains(event.target as Node)) {
        onClose();
      }
    };
    const handleEscape = (event: KeyboardEvent) => {
      if (event.key === 'Escape') onClose();
    };
    document.addEventListener('mousedown', handleClickOutside);
    document.addEventListener('keydown', handleEscape);
    return () => {
      document.removeEventListener('mousedown', handleClickOutside);
      document.removeEventListener('keydown', handleEscape);
    };
  }, [onClose]);
  const updateField = <K extends keyof UIPreferences>(key: K, value: UIPreferences[K]) => {
    const next = { ...preferences, [key]: value };
    onUpdate(next);
    saveUIPreferences(next);
  };
  return (
    <AnimatePresence>
      <motion.div 
        ref={panelRef}
        initial={{ opacity: 0, y: 4 }}
        animate={{ opacity: 1, y: 0 }}
        exit={{ opacity: 0, y: 4 }}
        transition={{ duration: 0.15, ease: "easeOut" }}
        className="absolute top-12 right-0 w-[300px] border border-[var(--ui-border)] rounded-[16px] shadow-[0_12px_40px_rgba(0,0,0,0.3)] z-[100] flex flex-col overflow-hidden bg-[var(--ui-surface-raised)]"
      >
        <header className="p-5 flex items-center justify-between">
          <h3 className="text-[var(--ui-text-primary)] font-bold text-[14px] tracking-tight">ตั้งค่าการแสดงผล</h3>
          <button onClick={onClose} className="p-1 text-[var(--ui-text-muted)] hover:text-[var(--ui-text-primary)] transition-colors">
            <span className="material-symbols-outlined text-[20px] font-bold">close</span>
          </button>
        </header>
        <div className="px-5 pb-5 space-y-5">
          <SettingSection label="ขนาดตัวอักษร">
            <SegmentedControl 
              options={[{ label: 'เล็ก', value: 'small' }, { label: 'กลาง', value: 'medium' }, { label: 'ใหญ่', value: 'large' }]}
              value={preferences.textSize}
              onChange={(v) => updateField('textSize', v as TextSize)}
            />
          </SettingSection>
          <SettingSection label="ขนาดการ์ด">
            <SegmentedControl 
              options={[{ label: 'S', value: 'S' }, { label: 'M', value: 'M' }, { label: 'L', value: 'L' }]}
              value={preferences.cardSize}
              onChange={(v) => updateField('cardSize', v as CardSize)}
            />
          </SettingSection>
          <SettingSection label="ธีมแสดงผล">
            <SegmentedControl 
              options={[{ label: 'สว่าง', value: 'light' }, { label: 'มืด', value: 'dark' }]}
              value={preferences.theme}
              onChange={(v) => updateField('theme', v as UITheme)}
            />
          </SettingSection>
          <div className="pt-2">
            <button 
              onClick={() => {
                onUpdate(DEFAULT_UI_PREFERENCES);
                saveUIPreferences(DEFAULT_UI_PREFERENCES);
              }} 
              className="text-[var(--ui-text-muted)] hover:text-[var(--ui-text-primary)] font-bold transition-colors text-[11px] uppercase tracking-widest"
            >
               คืนค่าเริ่มต้น
            </button>
          </div>
        </div>
      </motion.div>
    </AnimatePresence>
  );
};
const SettingSection: React.FC<{ label: string; children: React.ReactNode }> = ({ label, children }) => (
  <div className="space-y-1.5">
    <label className="text-[var(--ui-text-muted)] text-[10px] font-extrabold uppercase tracking-widest block pl-0.5">{label}</label>
    {children}
  </div>
);
const SegmentedControl: React.FC<{ options: { label: string; value: string }[]; value: string; onChange: (val: string) => void; }> = ({ options, value, onChange }) => (
  <div className="flex bg-[var(--ui-bg)]/40 p-0.5 rounded-lg border border-[var(--ui-border)] gap-0.5 shadow-inner">
    {options.map(opt => (
      <button
        key={opt.value}
        onClick={() => onChange(opt.value)}
        className={`flex-1 py-1.5 rounded-md text-[11px] font-bold uppercase transition-all duration-200 ${
          value === opt.value 
            ? 'bg-[var(--ui-primary)] text-white shadow-sm' 
            : 'text-[var(--ui-text-muted)] hover:text-[var(--ui-text-secondary)]'
        }`}
      >
        {opt.label}
      </button>
    ))}
  </div>
);
