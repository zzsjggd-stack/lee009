const CustomConfigTextarea: React.FC<{
  value: string;
  onChange: (val: string) => void;
  onBlur: (e: React.FocusEvent<HTMLTextAreaElement>) => void;
  placeholder: string;
  isTranslating: boolean;
  translatingLabel: string;
}> = ({ value, onChange, onBlur, placeholder, isTranslating, translatingLabel }) => {
  const [isFocused, setIsFocused] = useState(false);
  const textareaRef = useRef<HTMLTextAreaElement>(null);
  const adjustHeight = () => {
    const el = textareaRef.current;
    if (!el) return;
    
    if (!isFocused) {
      el.style.height = '44px';
      el.style.overflowY = 'hidden';
      return;
    }
    el.style.height = 'auto';
    const scrollHeight = el.scrollHeight;
    const nextHeight = Math.min(Math.max(scrollHeight, 110), 220);
    el.style.height = `${nextHeight}px`;
    el.style.overflowY = scrollHeight > 220 ? 'auto' : 'hidden';
  };
  useEffect(() => {
    adjustHeight();
  }, [value, isFocused]);
  return (
    <div className="relative">
      <textarea
        ref={textareaRef}
        value={value}
        onChange={(e) => onChange(e.target.value)}
        onFocus={() => setIsFocused(true)}
        onBlur={(e) => {
          setIsFocused(false);
          onBlur(e);
        }}
        placeholder={placeholder}
        disabled={isTranslating}
        style={{ 
          height: '44px',
          transition: 'height 180ms ease, border-color 150ms ease, box-shadow 150ms ease'
        }}
        className={`w-full bg-[var(--ui-input)] border border-[var(--ui-border)] rounded-xl p-3 outline-none resize-none font-medium text-[var(--ui-text-primary)] transition-all focus:border-[var(--ui-accent)] ${isTranslating ? 'opacity-30' : 'opacity-100'} ${!isFocused ? 'overflow-hidden' : ''}`}
      />
      {isTranslating && (
        <div className="absolute inset-0 flex items-center justify-center gap-2 text-[var(--ui-accent)] font-bold text-[11px] uppercase tracking-widest bg-black/20 rounded-xl backdrop-blur-sm z-10">
          <span className="material-symbols-outlined text-sm animate-spin">sync</span>
          {translatingLabel}
        </div>
      )}
    </div>
  );
};
