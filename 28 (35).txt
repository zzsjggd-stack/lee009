import { getVideoModelDurations } from './constants';
/**
 * ประมาณเวลาพูด: ไทย ~12 ตัวอักษร/วิ, อังกฤษ ~2.5 คำ/วิ, จุดหยุด +0.3 วิ
 */
export const speechSeconds = (text?: string) => {
  const t = (text || '').replace(/^\s*[^:：\n]{1,20}[:：]\s*/gm, '').trim();
  if (!t) return 0;
  const thai = (t.match(/[\u0E00-\u0E7F]/g) || []).length;
  const words = t.replace(/[\u0E00-\u0E7F]/g, ' ').trim().split(/\s+/).filter(Boolean).length;
  const pauses = (t.match(/\.\.\.|[?!—]/g) || []).length;
  return thai / 12 + words / 2.5 + pauses * 0.3;
};
/**
 * คำนวณเวลาพูดสูงสุดที่คลิปความยาว d วินาทีจะรับไหว
 */
export const maxSpeechFor = (d: number) => d <= 4 ? 2 : d <= 6 ? 3.5 : d <= 8 ? 5 : 7;
/**
 * เลือกความยาวคลิปที่เหมาะสมกับเนื้อหาช็อตและโมเดลที่ใช้
 */
export const pickClipDuration = (shot: { dialogue?: string; shotKind?: string }, model: string) => {
  const allowed = getVideoModelDurations(model);
  const sp = speechSeconds(shot.dialogue);
  const want = sp > 0 ? (sp <= 2 ? 4 : sp <= 3.5 ? 6 : sp <= 5 ? 8 : 10) : (shot.shotKind === 'reaction' ? 4 : 6);
  // หาความยาวที่สั้นที่สุดที่พอเพียง แต่ไม่เกินความสามารถโมเดล
  return allowed.find(d => d >= want) ?? allowed[allowed.length - 1];
};
/**
 * จำนวนคลิปตามเวลาเรื่อง: เวลาใช้จริงต่อคลิป = ความยาวคลิป - เวลาที่เผื่อไว้ตัดต่อ
 */
export const clipsForDuration = (durationSeconds: number, clipSeconds: number, margin = 1) =>
  Math.max(1, Math.ceil((durationSeconds || 0) / Math.max(1, clipSeconds - margin)));
/**
 * แบ่งช่วงวินาทีในคลิป (คู่มือ Veo 3.1: timestamp prompting)
 * needs = วินาทีขั้นต่ำของแต่ละช่วง · margin = เผื่อไว้ตัดต่อ: 1 → ท้ายคลิปเงียบ 1 วิ, 2 → หัว 1 วิ + ท้าย 1 วิ
 * เวลาที่เหลือแจกให้ทุกช่วง · ใช้วินาทีเต็มทั้งหมด
 */
export const planBeatTimes = (needs: number[], clipSeconds: number, margin = 1) => {
  const total = Math.max(1, Math.round(clipSeconds));
  const need = needs.map(n => Math.max(1, Math.ceil(n)));
  let head = margin >= 2 ? 1 : 0;
  let tail = margin >= 1 ? 1 : 0;
  let used = head + tail + need.reduce((a, b) => a + b, 0);
  if (used > total && head) { head = 0; used--; }
  if (used > total && tail) { tail = 0; used--; }
  while (used > total) {
    let k = -1;
    for (let i = 0; i < need.length; i++) if (need[i] > 1 && (k < 0 || need[i] > need[k])) k = i;
    if (k < 0) break;
    need[k]--; used--;
  }
  let spare = Math.max(0, total - used);
  for (let i = 0; spare > 0 && need.length > 0; i = (i + 1) % need.length) { need[i]++; spare--; }
  const end = total - tail;
  const spans: [number, number][] = [];
  let t = head;
  for (const n of need) {
    const s = Math.min(t, Math.max(0, end - 1));
    const e = Math.min(end, Math.max(s + 1, t + n));
    spans.push([s, e]);
    t = e;
  }
  return { head: head ? [0, head] as [number, number] : null, spans, tail: tail ? [end, total] as [number, number] : null };
};
/** รูปแบบเวลาในพรอมป์ เช่น [00:02-00:05] */
export const fmtBeat = (span: [number, number]) => `[00:${String(span[0]).padStart(2, '0')}-00:${String(span[1]).padStart(2, '0')}]`;
/** ช่วงวินาทีของ beats ในคลิป — ใช้ร่วมกัน STEP 4 และ STEP 5 ให้เวลาตรงกัน */
export const beatTimes = (beats: { say?: string }[], clipSeconds: number, margin = 1) =>
  planBeatTimes(beats.map(b => (b.say ? speechSeconds(b.say) + 0.4 : 1)), clipSeconds, margin);
