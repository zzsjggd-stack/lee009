import { Asset, ConfigData, Shot } from '../types';
// ใช้เช็กว่าภาพชีทยังตรงกับข้อมูลล่าสุดไหม
const styleKey = (c: ConfigData) => c.style === 'Custom' ? `Custom:${c.customStyle || ''}` : c.style;
const toneKey = (c: ConfigData) => c.colorTone === 'Custom' ? `Custom:${c.customColorTone || ''}` : c.colorTone;
export const assetImageKey = (a: Asset, c: ConfigData) => [
  a.visualPrompt || '',
  styleKey(c),
  a.type === 'Location' ? toneKey(c) : '',
  (a.sourceReferences || []).map(r => r.mediaId).join(',')
].join('||');
export const isAssetImageOld = (a: Asset, c: ConfigData) => {
  if (!a.referenceBase64 && !a.referenceMediaId) return false;
  if (a.imageKey) return a.imageKey !== assetImageKey(a, c);
  return !!a.imageVisualPrompt && a.imageVisualPrompt !== a.visualPrompt;
};
// ใช้เช็กว่าวิดีโอยังตรงกับบทพูด/การเคลื่อนกล้องล่าสุดไหม
export const videoContentKey = (s: Shot) => [
  s.description || '',
  s.dialogue || '',
  s.moveId || '',
  s.moveCustom || '',
  s.speakerId || '',
  s.offScreenSpeaker ? '1' : '',
  ...(s.beats?.length ? [JSON.stringify(s.beats)] : [])
].join('||');
// ใช้ร่วมกัน STEP 4 และ STEP 5: เช็กว่าภาพช็อตยังตรงกับข้อมูลล่าสุดไหม
export const shotFrameKey = (s: Shot, c: ConfigData) => [
  [...(s.characterIds || [])].sort().join(','),
  s.locationId || '',
  [...(s.propIds || [])].sort().join(','),
  s.angleId || '',
  s.angleCustom || '',
  s.description || '',
  c?.style || '',
  c?.colorTone || '',
  ...(s.positions ? [s.positions] : []),
  ...(s.beats?.length ? [`${s.beats[0].cam}|${s.beats[0].act || ''}`] : [])
].join('||');
export const shotRefAssets = (s: Shot, assets: Asset[]): Asset[] => {
  const speaker = assets.find(a => a.id === s.speakerId && a.type === 'Character');
  const listed = assets.filter(a => a.type === 'Character' && s.characterIds?.includes(a.id));
  let inFrame = s.offScreenSpeaker ? listed.filter(a => a.id !== s.speakerId) : listed;
  if (speaker && !s.offScreenSpeaker && s.shotKind === 'dialogue' && !inFrame.some(a => a.id === speaker.id)) inFrame = [speaker, ...inFrame];
  const sp = inFrame.find(a => a.id === s.speakerId);
  const chars = sp ? [sp, ...inFrame.filter(a => a.id !== sp.id)] : inFrame;
  const location = assets.find(a => a.id === s.locationId);
  const props = assets.filter(a => s.propIds?.includes(a.id));
  return [
    ...chars.filter(a => !!a.referenceMediaId),
    ...(location?.referenceMediaId ? [location] : []),
    ...props.filter(a => !!a.referenceMediaId).slice(0, 3)
  ].slice(0, 10);
};
export const shotFrameStaleReason = (s: Shot, assets: Asset[], c: ConfigData): 'content' | 'refs' | null => {
  if (!s.firstFrameMediaId && !s.firstFrameBase64) return null;
  if (s.cameraChangedAfterImage || (!!s.frameSourceKey && s.frameSourceKey !== shotFrameKey(s, c))) return 'content';
  if (s.frameRefIds !== undefined && s.frameRefIds !== shotRefAssets(s, assets).map(a => a.referenceMediaId).join(',')) return 'refs';
  return null;
};
